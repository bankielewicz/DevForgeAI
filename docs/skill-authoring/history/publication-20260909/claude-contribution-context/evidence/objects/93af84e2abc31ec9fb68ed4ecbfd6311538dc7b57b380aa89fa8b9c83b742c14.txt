def run(name,command,expected_exit,expected_text):
 expectation={'name':name,'command':command,'expected_exit_status':expected_exit,'required_stdout_text':expected_text}
 with (outputs/(name+'-expectation.json')).open('x') as handle:json.dump(expectation,handle,indent=2);handle.write('\n')

         'assertions_pass':p.returncode==expected_exit and expected_text in p.stdout}
 with (outputs/(name+'.json')).open('x') as handle:json.dump(record,handle,indent=2);handle.write('\n')
 records.append(record)

  run('receipt-legitimate-verification-only-new',['python3','-B',str(helpers/'new-receipt.py'),'--file',str(file),'--expected-sha256',sha(file)],0,'overall=PASS')
