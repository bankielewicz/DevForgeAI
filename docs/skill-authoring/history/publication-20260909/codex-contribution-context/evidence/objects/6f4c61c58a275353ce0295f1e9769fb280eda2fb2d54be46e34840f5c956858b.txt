import hashlib,json,re,subprocess,sys
R=Path(__file__).resolve().parent; W=R.parents[4]; S=W/'project-experts/codex/devforgeai-contribution-context'; E=S/'evals'
H=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def inside(root,rel):
 p=(root/rel).resolve()
 if not p.is_relative_to(root.resolve()): raise ValueError('path escapes root: '+rel)
 if not p.is_file(): raise ValueError('missing file: '+rel)
 return p
def digest(p,expected):
 if H(p)!=expected: raise ValueError('digest mismatch: '+str(p))
def checks():
 selected=json.loads((R/'selected-inputs.json').read_text())
 for p,d in selected['files_sha256'].items():digest(Path(p),d)

  if p.is_file():
   assert '__pycache__' not in p.parts and p.suffix!='.pyc'

   assert body.endswith('\n'),p
   assert all(line==line.rstrip() for line in body.splitlines()),p
   if p.suffix=='.md' and not p.is_relative_to(E):
    for link in re.findall(r'\[[^\]]*\]\(([^)]+)\)',body):
     if '://' not in link and not link.startswith('#'):
      target=inside(p.parent,link.split('#')[0]) if not link.startswith('../') else (p.parent/link.split('#')[0]).resolve()
      assert target.is_relative_to(S) and target.is_file(),(p,link)
   if p.name=='SKILL.md' and p==S/'SKILL.md': assert '{{' not in body and '[TODO:' not in body
 d=json.loads((S/'references/derivation.json').read_text())['template']
 source=Path('/home/bryan/Projects/DevForge/framework/DevForge/.poc')/d['preserved_locator']
 digest(source,d['source_sha256']);digest(inside(S,d['destination']),d['destination_sha256'])
 assert source.read_bytes()==(S/d['destination']).read_bytes();assert d['refresh_triggers']
 ev=json.loads((E/'evals.json').read_text());cov=json.loads((E/'coverage.json').read_text())

  root=E/row['fixture_root'];packet=json.loads((root/'packet.json').read_text())

  if row['mode']=='orient':assert 'report_authority' not in packet
  else:assert packet['report_authority']['output']=='outbox/HANDOFF-next.md'
  if row['case_id']=='CCX-06':assert not (root/packet['required_unavailable_reference']['path']).exists()
 a=E/'fixtures/CCX-08-unchanged-resume'; b=E/'fixtures/CCX-08-explicit-transfer'

 # Discriminating mechanical checks, deliberately invalid inputs; no prose grading.
