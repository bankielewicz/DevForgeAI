#!/usr/bin/env python3
"""Terminal receipt checker, revision 3 (CC7-002, CC8-PROP-003).

Revision 1 asked two independent questions -- is this file's digest anywhere in
the response, and is some accepted path substring anywhere in the response --
and passed when both were true. Nothing tied the two together, so it passed a
swapped pair (a.md quoted with b.md's hash), an unrelated file elsewhere whose
hash happened to match, and a path under a root nobody selected that merely
ended in the accepted relative suffix.

A receipt is an association, not two coincidences. So this revision:

- takes the accepted locator forms from a declared mapping, fixed before
  measurement rather than widened after seeing output;
- matches a locator at token boundaries, so `/some/other/root/docs/x/IDEAS.md`
  is not an occurrence of `docs/x/IDEAS.md`;
- reads receipts as records in declared shapes, so a file's digest is the one
  written in a record with that file's locator (revision 3, CP5-CK-003: ranking
  candidates by distance let a preceding file's digest own the next locator once
  the locators were long enough);
- recomputes from current bytes, so a hash taken before a later write fails;
- names unsupported forms instead of falling through to a substring pass.

Revision 3 closes the two defects CC7-002 and CC8-PROP-003 name:

- `locator_occurrences` validated where a locator legitimately appears and then
  `records_for` threw that away, rescanning the whole response for the same
  string. So a valid bare locator somewhere in the text made `locs` nonempty,
  and the record match was then free to land on `/some/other/root/<suffix>` --
  the very occurrence the validation had just excluded. The validated spans are
  now carried into record matching: a record counts only when its locator
  begins at an offset the form check already accepted.
- `[0-9a-f]{64}` matched a 64-character window inside a longer hex run, so a
  65-character token was read as a valid digest. A digest token is now whole:
  no hex character may sit on either side of it.
"""
import argparse, hashlib, json, os, re, sys

HEX64 = re.compile(r"\b[0-9a-f]{64}\b")
ABBREV = re.compile(r"\b[0-9a-f]{4,}\s*(?:\.{2,}|…)\s*[0-9a-f]*")
SEP = "/"
WORDISH = set("abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789._-")
# CC10-RR-002. A locator is one complete path token or it is not this file's
# locator. WORDISH covered the obvious filename characters and missed the rest,
# so `/other+/work/arm/a.md` and `/work/arm/a.md+backup` -- two different
# complete paths -- were accepted as occurrences of `/work/arm/a.md` merely
# because `+` was not in the set. Anything a filesystem will accept inside a
# path component belongs here; the delimiters that legitimately wrap a locator
# in prose or Markdown (space, backtick, asterisk, pipe, quotes, parentheses,
# brackets, comma, semicolon, colon, newline) deliberately do not -- a Markdown
# link writes `](/work/arm/a.md)` and a sentence writes `/work/arm/a.md, and`,
# and both of those are still this file's locator. Only characters that would
# make it a *different filename* are boundaries.
PATH_CHAR = WORDISH | set("+~@%#=")


def sha256_file(p):
    h = hashlib.sha256()
    with open(p, "rb") as fh:
        for blk in iter(lambda: fh.read(1 << 16), b""):
            h.update(blk)
    return h.hexdigest()


def locator_occurrences(text, locator, form):
    """Occurrences of `locator` as a whole path token.

    A locator preceded by a separator is the tail of some other absolute path --
    possibly under a root nobody selected -- and is not an occurrence of this
    file. That is true of an absolute locator as much as a relative one: a
    receipt must own the selected project or sandbox locator, not sit at the end
    of somebody else's path (CC9-RR-002).
    """
    out = []
    for m in re.finditer(re.escape(locator), text):
        s, e = m.start(), m.end()
        nxt = text[e] if e < len(text) else ""
        if nxt in PATH_CHAR or nxt == SEP:
            continue
        prev = text[s - 1] if s > 0 else ""
        if prev in PATH_CHAR:
            continue
        # CC9-RR-002. Revision 3 applied this only to the relative form, on the
        # reasoning that an absolute locator carries its own root and therefore
        # cannot be a tail. It can: `/other//work/arm/a.md` contains
        # `/work/arm/a.md` starting at a separator, so the selected sandbox root
        # was accepted as a suffix of a root nobody selected. Ownership of the
        # locator is the criterion for every accepted form, so the guard is the
        # same one for all of them -- a locator preceded by a separator is the
        # tail of a longer path, and the tail of another path is not this file.
        if prev == SEP:
            continue
        out.append((s, e))
    return out


def accepted_locators(promise, mapping):
    """Locator strings this delivery may legitimately use, from the declared map."""
    got = []
    rel = promise["label"]
    for form in mapping["accepted_forms"]:
        if form == "project_relative":
            got.append((rel, form))
        elif form == "sandbox_absolute":
            root = mapping.get("sandbox_root")
            if root:
                got.append((root.rstrip(SEP) + SEP + rel, form))
        elif form == "host_absolute":
            got.append((promise["path"], form))
        else:
            got.append((None, form))
    return got


# The glue between a locator and its digest is real prose -- a backtick, an
# em dash, a "SHA-256" label, a table pipe. Enumerating punctuation misses the
# labels; allowing anything would let a distant digest drift back into range.
# So glue is a bounded run that provably contains no other digest, and, for the
# same-line shapes, no newline: a newline is what separates one line-oriented
# receipt from the next, and crossing one is how the previous file's digest
# reached the following locator in the first place.
GLUE = r"(?:(?![0-9a-f]{{64}})[^\n]){{0,40}}"

# A digest token is the whole run or it is not a digest. Revision 2 wrote a bare
# {{64}} quantifier, which happily matched the first 64 characters of a 65-character
# run and reported a well-formed receipt for a token that is not a SHA-256 at all.
# The guards on both sides make the match refuse any longer hex run instead of
# taking a window out of it.
DIGEST = r"(?<![0-9a-f])(?P<digest>[0-9a-f]{{64}})(?![0-9a-f])"

# The locator carries its own group so a match can be checked against the spans
# the form validation already accepted, rather than against the raw string.
RECORD_SHAPES = {
    "locator_then_digest": r"(?P<loc>{loc})" + GLUE + DIGEST,
    "digest_then_locator": DIGEST + GLUE + r"(?P<loc>{loc})",
    "labelled_next_line":
        r"(?P<loc>{loc})[ \t]*\r?\n" + GLUE + DIGEST,
}


# Two or more path components -- enough to be a locator someone could have meant,
# rather than a bare word. Used only to detect a *different* path standing
# between a locator and a digest.
OTHER_PATH = re.compile(r"(?:/[A-Za-z0-9._+~@%-]+){2,}/?")


def _between(text, m):
    """The text between the locator and the digest of one matched record."""
    ls, le = m.start("loc"), m.end("loc")
    ds, de = m.start("digest"), m.end("digest")
    return text[le:ds] if le <= ds else text[de:ls]


def records_for(text, locator, shapes, valid_starts):
    """Receipt records in this text that name `locator` at a validated offset.

    Revision 2 asked which digest was *nearest* the locator, with no other
    promised locator in between. Distance is not ownership: once the locators
    were long enough, the digest written for the previous file sat closer to
    the next locator than that file's own digest did, and the second half of a
    swapped pair passed (CP5-CK-003). So this revision stops ranking candidates
    and reads records instead -- a locator and a digest written together, in a
    shape declared before measurement. A digest that is not in a record with
    this locator is not this file's receipt, however close it happens to be.

    Two records naming the same locator with different digests are not resolved
    by preferring one. That syntax is ambiguous for this file and is rejected.

    `valid_starts` is the set of offsets at which `locator_occurrences` accepted
    this string as a whole path token for this file. Revision 2 rescanned the
    response here and lost that work: a relative locator embedded in an
    unselected absolute root is a substring match like any other, so once some
    other occurrence had made the form check pass, a record could be read out of
    the very path the check had rejected (CC7-002). A match whose locator does
    not begin at an already-validated offset is not this file's locator.
    """
    out = []
    for name in shapes:
        pat = RECORD_SHAPES.get(name)
        if pat is None:
            out.append({"shape": name, "unsupported": True})
            continue
        rx = pat.format(loc=re.escape(locator))
        for m in re.finditer(rx, text, re.I):
            if m.start("loc") not in valid_starts:
                # The form check already rejected this occurrence -- typically a
                # relative suffix hanging off a root nobody selected. Reading a
                # digest out of it would restore exactly the bug this revision
                # closes.
                continue
            glue = _between(text, m)
            owner = OTHER_PATH.search(glue)
            if owner is not None:
                # CC10-RR-003. The digest is written against a different path
                # that sits between this locator and it. The form check already
                # refused that path -- typically a wrong root -- and refusing it
                # is exactly why its digest was left looking unowned, so the
                # labelled_next_line shape handed it to the valid bare locator
                # on the line above. A digest introduced by another path is that
                # path's digest, whether or not that path was accepted.
                out.append({"shape": name, "rejected": "INTERVENING_LOCATOR",
                            "intervening": owner.group(0),
                            "locator_span": (m.start("loc"), m.end("loc")),
                            "reason": "a different path token (%s) stands between this locator "
                                      "and the digest, so the digest is written against that "
                                      "path and is not this file's receipt"
                                      % owner.group(0)})
                continue
            out.append({"shape": name, "digest": m.group("digest").lower(),
                        "locator_span": (m.start("loc"), m.end("loc")),
                        "span": (m.start(), m.end())})
    return out


def check(promise, terminal, mapping):
    path, label = promise["path"], promise["label"]
    out = {"receipt_for": label, "path": path,
           "accepted_forms": list(mapping["accepted_forms"])}
    if not os.path.isfile(path):
        out.update(outcome="FAIL", reason="promised artifact does not exist at %s" % path)
        return out
    actual = sha256_file(path)
    out["actual_final_sha256"] = actual

    locs = []
    for text_loc, form in accepted_locators(promise, mapping):
        if text_loc is None:
            out.update(outcome="FAIL",
                       reason="declared locator form %r is not supported by this checker; "
                              "unsupported receipt syntax is rejected rather than matched "
                              "loosely" % form)
            return out
        for span in locator_occurrences(terminal, text_loc, form):
            locs.append({"form": form, "locator": text_loc, "span": span})
    out["locator_occurrences"] = [{"form": l["form"], "locator": l["locator"]} for l in locs]

    if not locs:
        m = ABBREV.search(terminal)
        if m:
            out["abbreviation_observed"] = m.group(0)
        out.update(outcome="FAIL", path_delivered=False,
                   reason="terminal response gives no accepted locator for this file; a path "
                          "under an unselected root that merely ends in the same relative "
                          "suffix is not this file")
        return out
    out["path_delivered"] = True

    shapes = mapping.get("record_shapes")
    if not shapes:
        out.update(outcome="FAIL", association="no declared record shape",
                   reason="the locator mapping declares no receipt record_shapes, so there is no "
                          "fixed syntax in which this file's digest could be said to belong to "
                          "its locator. Declare the shapes before measurement rather than "
                          "inferring ownership from proximity")
        return out
    out["record_shapes"] = list(shapes)

    # Group the validated offsets by locator string: one string may legitimately
    # occur several times, and every one of those offsets is admissible, while an
    # occurrence the form check dropped is admissible at none of them.
    valid_starts = {}
    for l in locs:
        valid_starts.setdefault(l["locator"], set()).add(l["span"][0])
    out["validated_locator_spans"] = {k: sorted(v) for k, v in valid_starts.items()}

    recs, unsupported = [], []
    for l in {l["locator"]: l for l in locs}.values():
        for r in records_for(terminal, l["locator"], shapes, valid_starts[l["locator"]]):
            if r.get("unsupported"):
                unsupported.append(r["shape"])
            else:
                recs.append(dict(r, form=l["form"], locator=l["locator"]))
    if unsupported:
        out.update(outcome="FAIL", association="unsupported record shape",
                   reason="declared record shape(s) %s are not implemented by this checker; an "
                          "unsupported receipt syntax is rejected by name rather than matched "
                          "loosely" % ", ".join(sorted(set(unsupported))))
        return out

    out["records"] = [{"shape": r["shape"], "locator": r["locator"], "digest": r["digest"]}
                      for r in recs]
    seen_digests = {r["digest"] for r in recs}

    if not recs:
        out.update(outcome="FAIL", association="locator outside any record",
                   reason="this file's locator appears in the response at a validated occurrence, "
                          "but never there in a declared receipt record shape paired with a whole "
                          "digest token. A digest written nearby, or one written against the same "
                          "relative suffix under a root nobody selected, is not a receipt for this "
                          "file")
        return out
    if len(seen_digests) > 1:
        out.update(outcome="FAIL", association="ambiguous",
                   reason="this locator appears in %d receipt records carrying %d different "
                          "digests (%s); the syntax does not say which is this file's, so it is "
                          "rejected rather than resolved by preferring one"
                          % (len(recs), len(seen_digests), ", ".join(sorted(seen_digests))))
        return out
    if seen_digests == {actual}:
        forms = sorted({r["form"] for r in recs})
        out.update(outcome="PASS", association=recs[0]["shape"],
                   reason="terminal response names this file by its %s locator in a declared "
                          "receipt record (%s) carrying all 64 characters of its own current "
                          "digest" % ("/".join(forms), recs[0]["shape"]))
        return out

    all_found = set(HEX64.findall(terminal))
    if actual in all_found:
        out.update(outcome="FAIL", association="unpaired",
                   reason="this file's digest occurs in the response but is not the digest paired "
                          "with its own locator; an unrelated occurrence elsewhere is not a "
                          "receipt for this file")
        return out
    if all_found:
        out["mismatched_digests"] = sorted(all_found)
        out.update(outcome="FAIL", association="wrong digest",
                   reason="the locator for this file is paired with a digest that is not its "
                          "own current bytes (%s); a hash taken before a later write, or a hash "
                          "belonging to another file, both land here" % actual)
        return out
    m = ABBREV.search(terminal)
    if m:
        out["abbreviation_observed"] = m.group(0)
        out.update(outcome="FAIL",
                   reason="the digest for this file is abbreviated (%r); an abbreviated hash "
                          "cannot be checked" % m.group(0))
        return out
    out.update(outcome="FAIL", reason="terminal response delivers no digest for this file")
    return out


def evaluate(cfg):
    src = cfg["terminal_source"]
    mapping = cfg["locator_mapping"]
    if not os.path.isfile(src):
        return {"schema_version": "devforge.terminal-receipt/v2", "conformance": "COULD_NOT_RUN",
                "reason": "terminal response not found at %s" % src}
    terminal = open(src, encoding="utf-8", errors="replace").read()
    # Revision 2 computed every other file's spans and attached them under
    # `_other_spans`, which nothing read. Ownership is decided inside check() from
    # this file's own validated spans and its own record, so the unused plumbing
    # is gone rather than left looking load-bearing.
    rows = [check(p, terminal, mapping) for p in cfg["promised"]]
    counts = {}
    for r in rows:
        counts[r["outcome"]] = counts.get(r["outcome"], 0) + 1
    return {"schema_version": "devforge.terminal-receipt/v2", "checker_revision": 3,
            "terminal_source": src,
            "locator_mapping": mapping,
            "terminal_source_note": "The final assistant message only. A full hash in an earlier "
                                    "tool result is not a delivered receipt.",
            "association_note": "Each promised file must be named by a declared locator form and "
                                "paired with its own current digest. Presence of both facts "
                                "separately is not an association.",
            "span_note": "A receipt record is read only at an occurrence the locator-form check "
                         "accepted. Rescanning the response for the same string re-admits the "
                         "unselected-root occurrence the check just excluded (CC7-002).",
            "digest_token_note": "A digest is a whole 64-character token. A 64-character window "
                                 "taken out of a longer hex run is not one (CC8-PROP-003).",
            "receipts": rows, "counts": counts,
            "conformance": "FAIL" if counts.get("FAIL") else "PASS"}


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--config", required=True)
    ap.add_argument("--out")
    a = ap.parse_args()
    result = evaluate(json.load(open(a.config)))
    txt = json.dumps(result, indent=1)
    if a.out:
        open(a.out, "w").write(txt + "\n")
    print(txt)
    return 0 if result["conformance"] == "PASS" else 1


if __name__ == "__main__":
    sys.exit(main())
