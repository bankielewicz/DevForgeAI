def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as fh:
        for b in iter(lambda: fh.read(1 << 16), b""):
            h.update(b)
    return h.hexdigest()


def tree(root):
    out = {}
    for d, _, fs in os.walk(root):
        for f in fs:
            p = os.path.join(d, f)
            out[os.path.relpath(p, root)] = sha(p)
    return out


def launch(arm_id, prompt, sandbox, staged, cap_assigned, extra_ro=()):
    d, why = CTL.claim_attempt(arm_id)
    if d is None:
        return {"arm": arm_id, "outcome": "COULD_NOT_RUN", "detail": why}
    cap, capdet = BUD.cap_for(cap_assigned)
    if cap < 60:
        return {"arm": arm_id, "outcome": "NOT_RUN",
                "detail": {"reason": "remaining overall budget below the minimum useful cap",
                           **capdet}}
    proj = os.path.join(d, "project")
    os.makedirs(proj, exist_ok=True)
    open(os.path.join(proj, "CLAUDE.md"), "w").write(CLAUDE_MD)
    staged_map = {}
    for src, dst in staged:
        s = src if os.path.isabs(src) else os.path.join(FIX, src)
        t = os.path.join(proj, dst)
        os.makedirs(os.path.dirname(t), exist_ok=True)
        shutil.copyfile(s, t)
        staged_map[dst] = {"from": s, "sha256": sha(t)}
    binds = H.client_state_binds(d) + ["--bind", proj, sandbox]
    for a, b in extra_ro:
        binds += ["--ro-bind", a, b]
    if EXPORT is None:
        # R10-CI-002. There is no authentic exported package to mount. R10 built
        # a command against an assumed export/export-01 directory, which would
        # have launched a worker against bytes nobody selected -- or against
        # nothing. An unavailable export is a dependent blocker, and reporting
        # it as one is the honest outcome; it is never fabricated and never
        # reported as an installation.
        return {"arm": arm_id, "outcome": "COULD_NOT_RUN", "launched": False,
                "marker_written": False,
                "detail": {"reason": "no authentic exported package is selected or available; "
                                     "refusing to construct a launch against an assumed export "
                                     "root",
                           "dependent_blockers": DEPENDENT_BLOCKERS}}
    cmd = H.build_cmd(sandbox, prompt, (EXPORT + "/skills/devforge-brainstorm",
                                        "/opt/devforgeai/skills/devforge-brainstorm"),
                      ALLOWED, binds)
    t0 = time.time()
    LAUNCH_MARKERS.append(arm_id)
    if LAUNCH_HOOK is None:
        obs = H.stream(cmd, os.path.join(d, "transcript.jsonl"), cap,
                       stderr_path=os.path.join(d, "stderr.log"))
    else:
        # A deterministic test drives this same function so that what is
        # measured is the real admission path, not a second implementation of
        # what it ought to do. The hook stands exactly where the model call is;
        # it records a marker and never reaches an executable.
        obs = LAUNCH_HOOK(arm_id, cmd, d, cap)
    rec = {"arm": arm_id, "run_id": FREEZE["run_id"], "session": FREEZE["session"],
           "sandbox_project": sandbox, "attempt_dir": d,
           "cap": capdet, "staged_inputs": staged_map,
           "prompt_sha256": hashlib.sha256(prompt.encode()).hexdigest(),
           "started_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime(t0)),
           "observation": obs, "outputs": tree(proj)}
    rec["outcome"] = "PASS" if obs["terminal"] == "COMPLETED" else (
        "COULD_NOT_RUN" if obs["terminal"] in ("TIMEOUT", "EOF_WITHOUT_RESULT") else "FAIL")
    if obs.get("result_text"):
        open(os.path.join(d, "terminal-response.txt"), "w").write(obs["result_text"])
    json.dump(rec, open(os.path.join(d, "arm.json"), "w"), indent=1)
    return rec

def a_arm(spec):
    """Phase A observes discovery, so it must not be told what to discover.

    C and B prepend HEAD, which names the installed SKILL.md and tells the
    worker to follow it. Prepending that here would answer A's own question --
    whether the skill is selected from the request alone -- so A sends the
    selector query unchanged and records activation as its own observation,
    separate from whether any artifact was later written.

    CHG-012/CHG-014: the launch/nonlaunch decision is a_arm.run_a_arm's, and the
    prompt actually handed to the launcher is passed to it so query identity is
    compared rather than asserted. A refusal returns here as a nonlaunch record
    and is never turned into an arm result.
    """
    q = A_QUERIES[spec["case"]]
    decision = AA.run_a_arm(spec["case"], q, BUD.remaining(), spec["assigned_cap_s"],
                            RUN, launcher=None, prompt=q["query"])
    if not decision.get("launched"):
        # Truthful nonlaunch: no marker, nothing observed about the candidate.
        decision["arm"] = spec["id"]
        decision["counts_as_arm_result"] = AA.nonlaunch_is_not_a_result(decision)
        return decision
    rec = launch(spec["id"], q["query"], "/work/%s" % spec["id"], [],
                 spec["assigned_cap_s"])
    rec["a_decision"] = decision
    rec["activation"] = _activation(rec, q)
    json.dump(rec, open(os.path.join(rec["attempt_dir"], "arm.json"), "w"), indent=1)
    return rec


def _activation(rec, q):
    """Was the skill actually selected, and was that the expected outcome?

    Revision 3 kept a private copy here that decided selection by asking whether
    one transcript line contained both `"devforge-brainstorm"` and `"Skill"` as
    substrings. Every session's opening metadata contains both, so it read
    availability and reported selection. control/activation.py is the corrected
    observation and it is what runs now; this wrapper exists only so the call
    site stays where the locator says it is (cases_r9: run_phases.py
    `_activation`).
    """
    return ACT.activation(rec, q)

def _completed_view(results, consumed):
    """What each finished arm actually established, in the shape the
    prerequisite predicate reads.

    C and B contribute a checked mechanical outcome and a bound judgment; A
    arms contribute their completion and their own activation observation,
    which is the distinction R9-CI-008 turns on.
    """
    view = {}
    for arm_id, rec in (results or {}).items():
        gpath = os.path.join(RUN, "grades", "%s-grade.json" % arm_id)
        mech = None
        if os.path.isfile(gpath):
            try:
                mech = json.load(open(gpath)).get("mechanical")
            except (ValueError, OSError):
                mech = None
        jpath = os.path.join(JUDGMENTS, "%s.json" % arm_id)
        view[arm_id] = {"native": rec.get("outcome"), "mechanical": mech,
                        "judgment": consumed.get(arm_id),
                        "judgment_bound_to": _fhash(jpath),
                        "activation": rec.get("activation")}
    return view


def _grade(arm_id, rec, plan):
    """Grade one completed arm now, before anything else is admitted.

    CHG-007: the preservation decision is made here, against live bytes at
    judgment time, and it is folded into the arm's mechanical outcome. A grade
    that ignored it left `preserved_outputs` decided by an inventory key.
    """
    gd = os.path.join(RUN, "grades")
    os.makedirs(gd, exist_ok=True)
    g = G.grade_arm(arm_id, rec, plan, gd)
    selected = _preservation_map(arm_id, rec, plan)
    g["preservation"] = AD.preserve_check(selected, os.path.join(rec["attempt_dir"], "project"))
    if g["preservation"]["outcome"] != "PASS":
        g["mechanical"] = "FAIL"
        g["mechanical_note"] = (g.get("mechanical_note", "") +
                                " Preservation failed: a selected preserved output is not holding "
                                "the bytes that were selected for it.")
    json.dump(g, open(os.path.join(gd, "%s-grade.json" % arm_id), "w"), indent=1)
    return g


def _preservation_map(arm_id, rec, plan):
    """The selected original digest for each of this arm's preserved outputs.

    The original is the digest the path had when it was staged, taken before the
    write that superseded it -- so it comes from the arm's own staged inputs,
    which are recorded bytes, not from the after-inventory the check is meant to
    be independent of. A preserved path with no staged original is left with a
    None original, which `preserve_check` cannot match and therefore refuses.
    """
    staged = rec.get("staged_inputs") or {}
    out = {}
    for rel in plan.get("outputs", {}).get(arm_id, {}).get("preserved_outputs", []):
        out[rel] = (staged.get(rel) or {}).get("sha256")
    return out


def _fhash(path):
    try:
        return hashlib.sha256(open(path, "rb").read()).hexdigest()
    except OSError:
        return None


def _criteria_revision(plan):
    """The fixed criteria, which are frozen and therefore cannot move.

    Criteria are an immutable input; judgments are generated evidence. Binding
    them at the same moment is what let a prefilled PASS look like an
    observation, so they are bound separately: this digest comes from the frozen
    plan, the judgment digest is taken when the gate consumes it.
    """
    return hashlib.sha256(
        json.dumps(plan.get("criteria"), sort_keys=True).encode()).hexdigest()


def _evidence_binding(arm_id, rec, grade, plan):
    """Exactly what a judgment for this arm has to name to be about this arm."""
    gd = os.path.join(RUN, "grades")
    b = {"run_id": FREEZE["run_id"], "session": FREEZE["session"], "arm": arm_id,
         "criteria_revision": _criteria_revision(plan),
         "result_sha256": _fhash(os.path.join(rec["attempt_dir"], "arm.json")),
         "output_inventory_sha256": hashlib.sha256(
             json.dumps(grade.get("inventory"), sort_keys=True).encode()).hexdigest()}
    for n in ("references", "labels", "receipts"):
        b["%s_sha256" % n] = _fhash(os.path.join(gd, "%s-%s.json" % (arm_id, n)))
    return b


def _judgment(arm_id, rec, grade, plan):
    """Consume an authorized judgment written after this arm produced output.

    Revision 2 loaded control/SEMANTIC-DECISIONS.json once, before C launched,
    and read that same in-memory object at every gate. A judgment written there
    in advance judged an arm whose output did not exist yet; one written
    afterwards changed a frozen input and was invisible to the running process.
    Either way nothing could be observed.

    So the judgment lives outside the freeze, in RUN/judgments/, and has to name
    this run, session and arm together with the digests of the completed result,
    the output inventory and all three checker outputs, under the frozen
    criteria revision. Those digests do not exist before the arm runs, which is
    what makes a prepopulated PASS fail here rather than pass silently. A
    missing judgment leaves downstream work unstarted; a stale or mismatched one
    stops it.
    """
    want = _evidence_binding(arm_id, rec, grade, plan)
    path = os.path.join(JUDGMENTS, "%s.json" % arm_id)
    base = {"arm": arm_id, "expected_binding": want, "judgment_path":
            os.path.relpath(path, RUN)}
    if not os.path.isfile(path):
        base.update(outcome="MISSING",
                    basis="no authorized judgment for this arm; absence cannot default to "
                          "PASS, and the gate leaves downstream work unstarted")
        return base
    try:
        j = json.load(open(path))
    except (ValueError, OSError) as exc:
        base.update(outcome="UNREADABLE", basis="judgment could not be parsed: %s" % exc)
        return base
    base["judgment_sha256"] = _fhash(path)
    missing = [k for k in want if k not in j]
    if missing:
        base.update(outcome="MALFORMED",
                    basis="judgment does not state %s, so it cannot be shown to be about this "
                          "arm's actual evidence" % ", ".join(sorted(missing)))
        return base
    bad = sorted(k for k, v in want.items() if str(j.get(k)) != str(v))
    if bad:
        base.update(outcome="STALE", mismatched=bad,
                    observed={k: j.get(k) for k in bad},
                    basis="judgment names different %s than this arm's current evidence; a "
                          "judgment about other bytes is not an observation about these"
                          % ", ".join(bad))
        return base
    if j.get("outcome") not in ("PASS", "FAIL"):
        base.update(outcome="MALFORMED",
                    basis="judgment outcome %r is not PASS or FAIL" % j.get("outcome"))
        return base
    # R10-CI-009. Everything above checks that the judgment *advertises* the
    # right digests. A dictionary carrying the published binding and
    # outcome PASS satisfied all of it without naming who judged, and without
    # any of the raw evidence that judgment was supposed to be about. Two
    # further things are required at the consumption boundary:
    #
    #   an identified authorized reviewer   -- a judgment is somebody's, and an
    #                                          unattributed one cannot be shown
    #                                          to be a post-output observation
    #   the actual evidence inventory       -- the raw output, checkpoint,
    #                                          runtime receipt, transcript and
    #                                          check bytes, present and matching
    #
    # Advertised hashes and an issued_at string do not establish chronology: a
    # string can be written at any time. Matching raw bytes that only exist
    # after the arm ran is what does.
    who = j.get("reviewer") or {}
    authorized = _authorized_reviewers()
    if not who.get("id"):
        base.update(outcome="UNATTRIBUTED",
                    basis="no reviewer identity; a judgment with no identified author cannot be "
                          "shown to be an authorized post-output judgment of this arm")
        return base
    if who["id"] not in authorized:
        base.update(outcome="UNAUTHORIZED_REVIEWER", reviewer=who.get("id"),
                    basis="reviewer %r is not in the selected authorized set %s"
                          % (who.get("id"), sorted(authorized)))
        return base
    ev = _raw_evidence_check(arm_id, rec, j)
    base["raw_evidence"] = ev
    if ev["outcome"] != "PASS":
        base.update(outcome="EVIDENCE_UNBOUND",
                    basis="the judgment does not bind the actual evidence it claims to judge: %s"
                          % ev["reason"])
        return base
    base.update(outcome=j["outcome"], criterion=j.get("criterion", "named behaviour"),
                basis=str(j.get("basis", ""))[:400], issued_at_utc=j.get("issued_at_utc"))
    return base


def _authorized_reviewers():
    """The reviewer identities this run will accept a judgment from.

    Selected, not inferred: an empty or absent selection authorizes nobody, so a
    run with no declared reviewers cannot consume any semantic judgment. That is
    the correct closed state, not a reason to accept anonymous ones.
    """
    p = os.path.join(RUN, "control", "AUTHORIZED-REVIEWERS.json")
    if not os.path.isfile(p):
        return set()
    try:
        return set(json.load(open(p)).get("authorized_reviewer_ids") or [])
    except (ValueError, OSError):
        return set()


RAW_EVIDENCE_KINDS = ("output_inventory", "checkpoint", "runtime_receipt", "transcript", "checks")


def _raw_evidence_check(arm_id, rec, j):
    """Did the reviewer actually have the bytes, and are they still those bytes?

    The judgment must list, per evidence kind, a path and the digest it judged.
    Each path is re-read here and re-hashed. A missing kind, an unreachable
    path or a changed digest all refuse: a judgment about other bytes -- or
    about no bytes -- is not an observation of this arm.
    """
    listed = j.get("evidence_inventory")
    if not isinstance(listed, dict):
        return {"outcome": "FAIL", "reason": "no evidence_inventory recorded",
                "required_kinds": list(RAW_EVIDENCE_KINDS)}
    missing = [k for k in RAW_EVIDENCE_KINDS if k not in listed]
    if missing:
        return {"outcome": "FAIL", "rows": [],
                "reason": "evidence_inventory is incomplete; missing %s" % ", ".join(missing)}
    rows, ok = [], True
    for kind in RAW_EVIDENCE_KINDS:
        entry = listed.get(kind) or {}
        path, want = entry.get("path"), entry.get("sha256")
        row = {"kind": kind, "path": path, "declared": want}
        if not path or not want:
            row.update(outcome="FAIL", reason="no path or digest recorded for this kind")
            ok = False
        else:
            full = path if os.path.isabs(path) else os.path.join(RUN, path)
            if not os.path.isfile(full):
                row.update(outcome="FAIL", reason="evidence file is not present at judgment "
                                                  "consumption")
                ok = False
            else:
                got = _fhash(full)
                row["actual"] = got
                if got != want:
                    row.update(outcome="FAIL", reason="evidence bytes changed since the judgment "
                                                      "named them")
                    ok = False
                else:
                    row.update(outcome="PASS")
        rows.append(row)
    return {"outcome": "PASS" if ok else "FAIL", "rows": rows,
            "reason": "" if ok else "; ".join(r["reason"] for r in rows if r.get("reason"))}


def _arm_gate(ctl, phase, arm_id, grade, sem, plan):
    """Bind this arm's checked results, by hash, into a phase decision."""
    gd = os.path.join(RUN, "grades")
    required = [{"id": "%s-%s" % (arm_id, k), "path": os.path.relpath(
                    os.path.join(gd, "%s-%s.json" % (arm_id, n)), RUN),
                 "field": "conformance", "expect": ["PASS"]}
                for k, n in (("references", "references"), ("labels", "labels"),
                             ("receipts", "receipts"))]
    required.append({"id": "%s-grade" % arm_id,
                     "path": os.path.relpath(os.path.join(gd, "%s-grade.json" % arm_id), RUN),
                     "field": "mechanical", "expect": ["PASS"]})
    return ctl.gate("%s:%s" % (phase, arm_id), required,
                    [{"case": arm_id, "criterion": sem.get("criterion", "named behaviour"),
                      "outcome": "PASS" if sem.get("outcome") == "PASS" else "FAIL",
                      "recorded": sem.get("outcome"),
                      "judgment_sha256": sem.get("judgment_sha256"),
                      "judgment_path": sem.get("judgment_path"),
                      "criteria_revision": (sem.get("expected_binding") or {}).get(
                          "criteria_revision"),
                      "basis": str(sem.get("basis"))[:200]}])


def main(argv=None):
    """C1, then B5, then B11, then A3a and A5a - each behind the last one's
    checks and its authorized post-output judgment, one arm at a time.

    Revision 1 launched C, checked only that the process completed, and
    submitted every B arm in the next statement. Revision 2 graded before
    admitting but kept a concurrent pool and a pre-loaded decision list, so a
    peer could finish ungraded and no judgment could ever be an observation.
    Revision 3 removes both: concurrency is 1, and the judgment is read after
    the output exists and must name that output's digests.
    """
    argv = sys.argv[1:] if argv is None else list(argv)
    resume = "--resume" in argv
    results, phase_state, consumed = {}, {}, {}
    paused = []

    def stop_here(reason, phase):
        CTL.failed = True
        phase_state[phase] = {"decision": "STOP", "reason": reason}

    def refuse(code, message, gate="startup"):
        print(message)
        CTL.decisions.append({"phase_gate": gate, "decision": "STOP", "reason": message})
        CTL.save(os.path.join(RUN, "control", "PHASES.json"))
        return code

    # Fail closed before anything can launch. A preparation freeze authorizes no
    # native call, so the honest response to being run is a clean refusal at the
    # boundary -- not an empty arm list that falls through the phases and prints
    # PROCEED because nothing failed.
    if int(FREEZE.get("native_arms_authorized", 0)) <= 0:
        return refuse(5, "NO NATIVE ARM AUTHORIZED - this freeze permits zero model calls; "
                         "refusing before any launch, stand-in or marker")
    if not FREEZE.get("arms"):
        return refuse(5, "NO ARMS DECLARED - nothing to admit; refusing rather than reporting "
                         "an empty run as a completed one")
    conc = FREEZE.get("concurrency", {})
    wrong = sorted(k for k, v in conc.items() if int(v) != 1)
    if wrong:
        return refuse(6, "SEQUENTIAL POLICY VIOLATED - concurrency for %s is not 1; this runner "
                         "implements one-arm-at-a-time admission and will not run a policy it "
                         "cannot enforce" % ", ".join(wrong))

    # Verify before reading. The plan is a bound file; parsing it first would
    # mean acting on bytes whose integrity has not been established yet.
    if not CTL.verify_freeze():
        return refuse(2, "FREEZE MISMATCH - no arm launched")
    try:
        plan = json.load(open(os.path.join(RUN, "control", "PLAN.json")))
        prefilled = json.load(open(os.path.join(RUN, "control", "SEMANTIC-DECISIONS.json")))
    except (ValueError, OSError) as exc:
        return refuse(4, "BOUND CONFIGURATION UNREADABLE - no arm launched: %s" % exc)
    if prefilled.get("decisions"):
        return refuse(7, "FROZEN JUDGMENTS PRESENT - the bound decisions file carries %d "
                         "prewritten judgment(s); a judgment fixed before its arm produced "
                         "output is not an observation of that output"
                      % len(prefilled["decisions"]))

    gates = json.load(open(os.path.join(RUN, "control", "GATE-INPUTS.json")))
    if not CTL.gate("deterministic", gates["deterministic"], []):
        print("DETERMINISTIC GATE FAILED - native phase NOT_RUN")
        CTL.save(os.path.join(RUN, "control", "PHASES.json"))
        return 3

    os.makedirs(JUDGMENTS, exist_ok=True)
    os.makedirs(os.path.join(RUN, "decisions"), exist_ok=True)

    def run_one(phase, spec, launcher):
        """Launch, complete, check, then consume this arm's own judgment.

        Nothing later is admitted until this returns, which is the whole of the
        sequential policy: there is no pool and therefore no ungraded peer.
        """
        if not CTL.verify_freeze():
            stop_here("freeze drift detected before admitting %s" % spec["id"], phase)
            return False
        if not CTL.may_launch():
            results.setdefault(spec["id"], {
                "arm": spec["id"], "outcome": "NOT_RUN",
                "detail": "overall budget exhausted before this arm was admitted"})
            stop_here("budget exhausted before %s" % spec["id"], phase)
            return False
        # CHG-008. The prerequisite refusal happens here, before the launch and
        # therefore before any admission marker exists. Revision 3 enforced the
        # order by the sequence of calls in this function, which is an ordering
        # only for as long as nobody calls them out of order, and a refusal
        # after the fact leaves a marker behind saying the arm was admitted.
        pre = AD.prerequisites(spec["id"], _completed_view(results, consumed),
                               diagnostic_scope=DIAGNOSTIC_C_ONLY)
        json.dump(pre, open(os.path.join(RUN, "decisions",
                                         "prereq-%s.json" % spec["id"]), "w"), indent=1)
        if not pre.get("admit"):
            results.setdefault(spec["id"], {
                "arm": spec["id"], "outcome": "NOT_RUN", "prerequisites": pre,
                "detail": pre.get("reason")})
            stop_here("prerequisites refused %s: %s" % (spec["id"], pre.get("reason")), phase)
            return False
        # Resume must not relaunch a completed arm. Its result already exists;
        # relaunching would spend a native call to reproduce evidence that is
        # already on disk, and the second run would not be the thing the pending
        # judgment was written about.
        prior = os.path.join(RUN, "attempts", spec["id"], "arm.json")
        if resume and os.path.isfile(prior):
            try:
                rec = json.load(open(prior))
                rec["resumed_without_relaunch"] = True
            except (ValueError, OSError) as exc:
                stop_here("recorded result for %s is unreadable on resume: %s"
                          % (spec["id"], exc), phase)
                return False
        else:
            rec = launcher(spec)
        results[rec["arm"]] = rec
        json.dump(results, open(os.path.join(RUN, "control", "ARMS.json"), "w"), indent=1)
        if rec.get("outcome") != "PASS":
            print("  ", rec["arm"], rec["outcome"], "- stopping further admission")
            stop_here("%s did not complete natively" % rec["arm"], phase)
            return False
        if phase == "A":
            act = rec.get("activation", {})
            ok = act.get("outcome") == "PASS"
            print("  ", rec["arm"], rec["outcome"], "| activation", act.get("outcome"))
            if not ok:
                stop_here("%s did not show the expected activation" % rec["arm"], phase)
            return ok
        g = _grade(rec["arm"], rec, plan)
        sem = _judgment(rec["arm"], rec, g, plan)
        consumed[rec["arm"]] = sem.get("outcome")

        # The pause. The judgment for this arm can only be written now: its
        # binding names the digests of the result, the inventory and the three
        # checker outputs, none of which existed before this point. So when it
        # is absent the runner stops and states exactly what a judgment would
        # have to say to be about this arm, rather than advancing on nothing or
        # accepting something written earlier.
        if sem.get("outcome") == "MISSING":
            req = os.path.join(JUDGMENTS, "%s.REQUIRED.json" % rec["arm"])
            json.dump({"schema_version": "devforge.judgment-request/v1",
                       "arm": rec["arm"], "phase": phase,
                       "required_binding": sem["expected_binding"],
                       "write_to": os.path.relpath(
                           os.path.join(JUDGMENTS, "%s.json" % rec["arm"]), RUN),
                       "then": "re-invoke this runner with --resume; the completed arm is "
                               "read from its recorded result and is not relaunched",
                       "outcome_values": ["PASS", "FAIL"],
                       "note": "These digests describe evidence that exists now. A judgment "
                               "written before this arm ran cannot carry them."},
                      open(req, "w"), indent=1)
            paused.append(rec["arm"])
            phase_state[phase] = {"decision": "AWAITING_JUDGMENT", "arm": rec["arm"]}
            print("  ", rec["arm"], rec["outcome"], "| mechanical", g["mechanical"],
                  "| judgment MISSING - paused, request written to",
                  os.path.relpath(req, RUN))
            return False

        ok = _arm_gate(CTL, phase, rec["arm"], g, sem, plan)
        print("  ", rec["arm"], rec["outcome"], "| mechanical", g["mechanical"],
              "| judgment", sem.get("outcome"), "|", "ADMIT-NEXT" if ok else "STOP")
        if not ok:
            stop_here("%s failed a required check or judgment" % rec["arm"], phase)
        return ok

    # ---- Phase C ---------------------------------------------------------
    cspecs = [a for a in FREEZE["arms"] if a["phase"] == "C"]
    c_ok = True
    for spec in cspecs:
        c_ok = run_one("C", spec, lambda sp: launch(
            sp["id"], C_PROMPT, "/work/tierC-%s" % sp["id"].lower(), [],
            sp["assigned_cap_s"]))
        if not c_ok:
            break
    phase_state.setdefault("C", {"decision": "PROCEED" if c_ok else "STOP"})

    # ---- Phase B: one at a time, behind C -------------------------------
    bspecs = [a for a in FREEZE["arms"] if a["phase"] == "B"]
    b_ok = c_ok
    for spec in bspecs:
        if not b_ok:
            results.setdefault(spec["id"], {
                "arm": spec["id"], "outcome": "NOT_RUN",
                "detail": "never admitted: an earlier required arm did not clear its gate, or "
                          "the overall budget was exhausted"})
            continue
        b_ok = run_one("B", spec, b_arm)
    phase_state.setdefault("B", {"decision": "STOP" if CTL.failed else "PROCEED"})

    # ---- A gate: a checked result and a bound judgment for every C and B --
    req_ids = [s["id"] for s in cspecs] + [s["id"] for s in bspecs]
    missing = [i for i in req_ids
               if results.get(i, {}).get("outcome") != "PASS"
               or not os.path.isfile(os.path.join(RUN, "grades", "%s-grade.json" % i))
               or not os.path.isfile(os.path.join(JUDGMENTS, "%s.json" % i))]
    a_inputs = [{"id": "%s-grade" % i, "path": "grades/%s-grade.json" % i,
                 "field": "mechanical", "expect": ["PASS"]} for i in req_ids]
    a_sem = [{"case": i, "criterion": "named behaviour",
              "outcome": "PASS" if consumed.get(i) == "PASS" else "FAIL",
              "judgment_sha256": _fhash(os.path.join(JUDGMENTS, "%s.json" % i)),
              "basis": "judgment consumed at this arm's own gate"} for i in req_ids]
    # CHG-009 / CHG-015. Native admission additionally requires the five fixed
    # runtime observations and a record bound to the current selected
    # allocation. Both fail closed: the observation classes the selected v1
    # interface exposes no collector for are deferred by name and block here,
    # which is why this gate cannot reach PASS for a native arm on this
    # interface. That refusal is the honest outcome, not a defect to route
    # around.
    obs = AD.observations(RUNTIME_OBSERVED, task_identity=FREEZE.get("run_id"))
    auth = AD.authority_bind(SELECTED_ALLOCATION,
                             {"session": FREEZE.get("session"),
                              "revision": (FREEZE.get("execution_ref") or {}).get("revision"),
                              "path": (FREEZE.get("execution_ref") or {}).get("path"),
                              "sha256": (FREEZE.get("execution_ref") or {}).get("sha256")})
    json.dump({"observations": obs, "authority_binding": auth},
              open(os.path.join(RUN, "decisions", "a-admission.json"), "w"), indent=1)
    a_sem.append({"case": "runtime_observations", "criterion": "CHG-009 five fixed classes",
                  "outcome": obs["outcome"], "blocking": obs["blocking"],
                  "basis": "runtime observation consumer; deferred classes block admission"})
    a_sem.append({"case": "authority_binding", "criterion": "CHG-015 current allocation",
                  "outcome": auth["outcome"],
                  "basis": "new evidence must carry the selected current allocation"})
    a_ok = (not missing) and CTL.gate("A", a_inputs, a_sem)
    if missing:
        CTL.decisions.append({"phase_gate": "A", "decision": "STOP",
                              "reason": "required arms without a checked result and a bound "
                                        "judgment: %s" % missing})
        CTL.failed = True
    phase_state["A"] = {"decision": "PROCEED" if a_ok else "STOP"}

    # ---- Phase A: actually launched, one at a time ----------------------
    aspecs = [a for a in FREEZE["arms"] if a["phase"] == "A"]
    a_running = a_ok
    for spec in aspecs:
        if not a_running:
            results.setdefault(spec["id"], {
                "arm": spec["id"], "outcome": "NOT_RUN",
                "detail": "A gate did not consume a checked PASS and a bound judgment for "
                          "every required C and B arm" if not a_ok else
                          "an earlier A arm did not show the expected activation"})
            continue
        a_running = run_one("A", spec, a_arm)

    json.dump(results, open(os.path.join(RUN, "control", "ARMS.json"), "w"), indent=1)
    json.dump({"phases": phase_state, "launch_markers": list(LAUNCH_MARKERS),
               "awaiting_judgment": paused, "resumed": resume,
               "policy": "sequential, concurrency 1 for C, B and A"},
              open(os.path.join(RUN, "control", "PHASE-STATE.json"), "w"), indent=1)
    CTL.save(os.path.join(RUN, "control", "PHASES.json"))
    print("phases:", {k: v["decision"] for k, v in phase_state.items()},
          "| launch markers:", LAUNCH_MARKERS)
    if paused:
        print("AWAITING JUDGMENT for:", ", ".join(paused))
        return 8
    return 0


if __name__ == "__main__":
    sys.exit(main())
