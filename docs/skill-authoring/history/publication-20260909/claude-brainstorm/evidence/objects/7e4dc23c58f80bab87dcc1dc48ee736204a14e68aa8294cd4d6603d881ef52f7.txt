"""Preparation freeze, revision 3 (CC7-001, CC8-PROP-002).

Revision 1 bound the checkers and the candidate but not the runner, not the
grader, and not the generated grading configurations -- so the code that decided
what to launch, and the criteria it decided against, were outside the thing that
was supposed to pin them. A freeze that omits the orchestrator does not pin the
orchestration.

This binds every file the phase path reads or imports: the runner, the grader,
the controller, the harness, every checker, the plan (roots, alias map, criteria,
caps, phase order), the semantic-decision record, the gate inputs, the candidate
package, the task prompts and fixtures. Full bytes are preserved beside the
manifest so a later reader can diff rather than trust. The freeze does not carry
its own digest.

Revision 3 makes the two binding boundaries explicit and checkable, because
revision 2 left them implicit and two different readings of the same freeze were
both defensible:

- *Fixed inputs* are bound here, by name, whether or not anything ever consumes
  them. A selected input that no arm reads is still a selected input, and
  mutating it after the freeze invalidates the fixed criteria exactly as
  mutating a consumed one does (CC7-001).
- *Generated evidence* -- judgments, grades, checker output, anything that
  cannot exist yet -- is not bound here and cannot be. It is bound at its own
  consumption boundary. So creating or changing a file under the run root that
  is not in the selected set is not a freeze violation, and treating it as one
  would forbid the post-output judgments the plan requires (CC8-PROP-002).

`selected_fixed_inputs` now records, per entry, why it is in the set and what is
expected to consume it, so "selected but not yet consumed" is a state a reader
can see rather than infer. `verify` reports the two cases under different names
instead of one undifferentiated mismatch list.
"""

def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as fh:
        for b in iter(lambda: fh.read(1 << 16), b""):
            h.update(b)
    return h.hexdigest()


def tree(root, keep=lambda p: True):
    out = {}
    for d, _, fs in os.walk(root):
        if "__pycache__" in d:
            continue
        for f in fs:
            if f.endswith(".pyc"):
                continue
            p = os.path.join(d, f)
            if keep(p):
                out[os.path.relpath(p, root)] = sha(p)
    return out

ALLOCATION_KEYS = ("session", "artifact_id", "revision", "path", "sha256")


def selected_allocation(path=None):
    """The current issued allocation, read from a selected input.

    R9-CI-006 / CHG-015. Revision 3 wrote `SESSION-001@5` and that revision's
    digest into this emitter as constants, so every freeze it produced -- for
    any later allocation -- claimed the identity of a finished one. A historical
    authority is an input to new work, never the identity of it.

    The identity is therefore read from control/ALLOCATION.json, which is itself
    a bound entry of the selected code set, and it is validated before it can be
    emitted: a missing file, a missing field or a placeholder is refused here
    rather than carried into a record. No session digest is hardcoded in this
    module, and none may be: a future issuance supplies its own through this
    file.
    """
    path = path or os.path.join(RUN, "control", "ALLOCATION.json")
    if not os.path.isfile(path):
        raise SystemExit("NO SELECTED ALLOCATION - %s is absent. The freeze emits the current "
                         "issued allocation and will not substitute a historical one." % path)
    try:
        alloc = json.load(open(path))
    except ValueError as exc:
        raise SystemExit("SELECTED ALLOCATION UNREADABLE - %s" % exc)
    missing = [k for k in ALLOCATION_KEYS if not alloc.get(k)]
    if missing:
        raise SystemExit("SELECTED ALLOCATION INCOMPLETE - missing %s; refusing to emit a record "
                         "with an unbound authority identity" % ", ".join(missing))
    if not (isinstance(alloc["sha256"], str) and len(alloc["sha256"]) == 64):
        raise SystemExit("SELECTED ALLOCATION DIGEST MALFORMED - %r is not a whole 64-character "
                         "digest" % (alloc["sha256"],))
    return alloc

def verify(freeze, run_root=RUN, observed_extra=()):
    """Check the selected set against current bytes.

    Two questions that revision 2 ran together are answered separately here.

    `bound_mismatches` is a freeze violation: an entry of the selected set whose
    bytes changed, vanished, or -- for an entry that was unresolved at freeze
    time -- appeared. Every entry counts, consumed or not.

    `unbound_changes` is not a violation. `observed_extra` may carry paths a
    caller noticed under the run root; each one that is not an entry of the
    selected set is reported as generated evidence, so a reader can see it was
    considered and correctly not treated as a broken freeze.
    """
    roots = freeze.get("path_roots", {})
    bound = freeze["bound_hashes"]
    mismatches, unresolved = [], []
    for rel, want in bound.items():
        p = rel
        if not os.path.isabs(p):
            pre = p.split("/", 1)[0]
            p = (os.path.join(roots[pre], p.split("/", 1)[1])
                 if pre in roots and "/" in p else os.path.join(run_root, p))
        got = sha(p) if os.path.isfile(p) else None
        if want is None:
            # Unresolved at freeze time. It stays unresolved: an admission input
            # that appears afterwards was not the one the criteria were fixed
            # against, and an absent one still authorizes nothing.
            unresolved.append({"path": rel, "appeared": got is not None, "actual": got})
            continue
        if got != want:
            mismatches.append({"path": rel, "declared": want, "actual": got,
                               "consumed_by": (freeze.get("selected_fixed_inputs", {})
                                               .get(rel, {}).get("consumed_by")),
                               "note": "selected fixed input; binding does not depend on whether "
                                       "anything consumed it"})
    unbound = [x for x in observed_extra if x not in bound]
    return {
        "check": "freeze_binding",
        "bound": len(bound),
        "bound_mismatches": mismatches,
        "unresolved_bindings": unresolved,
        "unbound_changes": [{"path": x, "classification": "generated evidence",
                             "freeze_violation": False,
                             "note": "not an entry of the selected set; bound at its own "
                                     "consumption boundary, not here"} for x in unbound],
        "outcome": "PASS" if not mismatches and not unresolved else "FAIL",
        "outcome_note": "Unbound changes never contribute to this outcome.",
    }


def main():
    fz = build()
    out = os.path.join(RUN, "control", "FREEZE.json")
    json.dump(fz, open(out, "w"), indent=1)
    # preserve full bytes of the bound code so a reviewer can diff, not just trust
    import shutil
    dest = os.path.join(RUN, "freeze-bytes")
    if os.path.isdir(dest):
        shutil.rmtree(dest)
    for rel in CODE:
        t = os.path.join(dest, rel)
        os.makedirs(os.path.dirname(t), exist_ok=True)
        shutil.copyfile(os.path.join(RUN, rel), t)
    print("bound %d hashes (%d code files), bytes preserved at freeze-bytes/"
          % (fz["bound_count"], len(CODE)))
    return 0


if __name__ == "__main__":
    sys.exit(main())
