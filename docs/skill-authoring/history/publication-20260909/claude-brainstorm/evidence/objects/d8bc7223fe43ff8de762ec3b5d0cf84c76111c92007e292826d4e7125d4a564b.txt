#!/usr/bin/env python3
"""A-arm launch/completion and nonlaunch outcomes (CC7-008, CC7-010).

CHG-012 and CHG-014, authored against the selected r7 baseline
`control/controller.py:81-93` (`claim_attempt`) and PLAN.json:28,90-110.
Authored, not executed: `launch` takes the launcher as an argument and this
allocation supplies none.

CC7-010: `claim_attempt` already refuses a preexisting destination, but its
caller went on as though a refusal were a launch. A refused arm must leave no
admission marker, must leave the existing bytes exactly as they were, and must
record a nonlaunch outcome that is not an arm result. An ordinary nonlaunch
refusal is a clean, truthful outcome -- it is not a successful arm, and it is
not a failure of the candidate either.

CC7-008: an A arm's result is an activation observation bound to its own case
and its own unchanged query bytes. A marker is not the observation.
"""
import hashlib, json, os

A_CASES = ("A3a", "A5a")
NONLAUNCH = ("REJECTED_PREEXISTING_DESTINATION", "REJECTED_INSUFFICIENT_CAP")


def _sentinel_state(path):
    """What is at the destination now, recorded before anything is decided.

    CHG-014: revision 2 recorded `os.path.getsize`. Equal sizes are not equal
    bytes, so a size map could not tell "left exactly as found" from "rewritten
    to the same length", which is precisely the claim the nonlaunch path has to
    support. Digests are recorded instead, and the size is kept beside them
    because it is cheap and it makes a truncation obvious to a reader.
    """
    if not os.path.isdir(path):
        return None
    out = {}
    for d, _, fs in os.walk(path):
        for f in sorted(fs):
            p = os.path.join(d, f)
            rel = os.path.relpath(p, path)
            try:
                h = hashlib.sha256()
                with open(p, "rb") as fh:
                    for b in iter(lambda: fh.read(1 << 16), b""):
                        h.update(b)
                out[rel] = {"sha256": h.hexdigest(), "bytes": os.path.getsize(p)}
            except OSError as exc:
                out[rel] = {"sha256": None, "bytes": None, "unreadable": str(exc)[:120]}
    return out


def sentinel_preserved(before, after):
    """Did the refusal leave the existing evidence exactly as it was?

    A nonlaunch claims preservation, so the claim is checked rather than
    asserted: same key set, same digest for every key. A size-only inventory
    could not have answered this (CC7-010).
    """
    if before is None and after is None:
        return {"check": "sentinel_preservation", "outcome": "NOT_APPLICABLE",
                "reason": "no destination existed before or after; nothing to preserve"}
    if before is None or after is None:
        return {"check": "sentinel_preservation", "outcome": "FAIL",
                "reason": "destination appeared or vanished across the refusal"}
    added = sorted(set(after) - set(before))
    removed = sorted(set(before) - set(after))
    changed = sorted(k for k in set(before) & set(after)
                     if before[k].get("sha256") != after[k].get("sha256"))
    ok = not (added or removed or changed)
    return {"check": "sentinel_preservation",
            "outcome": "PASS" if ok else "FAIL",
            "added": added, "removed": removed, "changed": changed,
            "compared": len(before),
            "reason": ("every preexisting file is present with the byte-identical digest it had "
                       "before the refusal" if ok else
                       "the refusal did not leave the destination as it found it")}


SELECTED_CASES_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "A-CASES.json")


def load_selected_cases(path=None):
    """The fixed case-to-row mapping. Absent selection authorizes no A case."""
    path = path or SELECTED_CASES_PATH
    if not os.path.isfile(path):
        return {}
    try:
        return json.load(open(path)).get("cases") or {}
    except (ValueError, OSError):
        return {}


def run_a_arm(case, selector, budget_remaining, cap, run_root, launcher=None,
               prompt=None, selected_cases=None):
    """Admit, or refuse and say so.

    Returns a record whose `marker_written` is the whole question CC7-010 turns
    on: a refused arm writes none, and its `outcome` is a nonlaunch status
    rather than PASS or FAIL, because nothing about the candidate was observed.
    """
    if case not in A_CASES:
        return {"case": case, "outcome": "FAIL", "marker_written": False,
                "reason": "unknown A case; the plan names exactly %s" % ", ".join(A_CASES)}

    # R10-CI-007. R10 derived the digest from the selector it was handed and then
    # compared it with itself, so the check could only ever agree: the wrong
    # case, a changed query and a changed should_trigger all passed as
    # "unchanged". The fixed expectation must come from somewhere the caller
    # cannot edit on the way in, so it comes from control/A-CASES.json -- the
    # selected rows, with the candidate trigger file's own digest recorded
    # beside them.
    fixed = (selected_cases or load_selected_cases()).get(case)
    if fixed is None:
        return {"case": case, "outcome": "FAIL", "marker_written": False,
                "reason": "case %r is not one of the selected A cases; a case the fixed selection "
                          "does not name is not an A arm" % case}
    query_text = selector.get("query")
    if not isinstance(query_text, str):
        return {"case": case, "outcome": "COULD_NOT_RUN", "marker_written": False,
                "query_id": selector.get("id"),
                "reason": "selected row carries no query text, so no query identity can be "
                          "derived and no prompt can be compared against it"}
    selected_sha = hashlib.sha256(query_text.encode("utf-8")).hexdigest()
    launch_prompt = query_text if prompt is None else prompt
    launched_sha = hashlib.sha256(launch_prompt.encode("utf-8")).hexdigest()

    # Cross-case: the row handed in must be the row this case names.
    if selector.get("id") != fixed["row_id"]:
        return {"case": case, "outcome": "FAIL", "marker_written": False, "launched": False,
                "control_completion": "CLEAN", "expected_row_id": fixed["row_id"],
                "supplied_row_id": selector.get("id"),
                "reason": "case %s is bound to row %r, but row %r was supplied; an A observation "
                          "made with another case's selector answers another question"
                          % (case, fixed["row_id"], selector.get("id"))}
    # Modified row: the supplied query must be the selected bytes.
    if selected_sha != fixed["query_sha256"]:
        return {"case": case, "outcome": "FAIL", "marker_written": False, "launched": False,
                "control_completion": "CLEAN", "expected_query_sha256": fixed["query_sha256"],
                "supplied_query_sha256": selected_sha,
                "reason": "the supplied query does not hash to the selected row's query bytes; "
                          "this is a modified row, not the selected case"}
    # Modified expectation: should_trigger is fixed by the selection.
    if bool(selector.get("should_trigger")) != bool(fixed["should_trigger"]):
        return {"case": case, "outcome": "FAIL", "marker_written": False, "launched": False,
                "control_completion": "CLEAN",
                "expected_should_trigger": fixed["should_trigger"],
                "supplied_should_trigger": selector.get("should_trigger"),
                "reason": "the supplied expectation differs from the selected row's fixed "
                          "should_trigger; changing the expectation to match an outcome is the "
                          "move phase A exists to prevent"}

    rec = {"case": case, "query_id": selector.get("id"),
           "selected_query_sha256": selected_sha,
           "launched_prompt_sha256": launched_sha,
           "query_identity_source": "derived from the selected row's own query bytes",
           "prompt_bytes_unchanged": launched_sha == selected_sha,
           "marker_written": False,
           "note": "The case selector is explicit. A5a is not invoked by an A3a scenario, and a "
                   "valid A3a scenario requires A5a non-invocation to be recorded as its own PASS."}

    if selector.get("query_sha256") and selector["query_sha256"] != selected_sha:
        rec.update(outcome="FAIL", launched=False, control_completion="CLEAN",
                   declared_query_sha256=selector["query_sha256"],
                   reason="the row declares a query digest that does not equal the digest of its "
                          "own query bytes; the declared value is not adopted")
        return rec
    if not rec["prompt_bytes_unchanged"]:
        rec.update(outcome="FAIL", launched=False, control_completion="CLEAN",
                   reason="the prompt handed to the launcher is not the selected query bytes. "
                          "Phase A observes selection from the unchanged selector, so a "
                          "launch-time mutation makes the observation answer a different question")
        return rec

    dest = os.path.join(run_root, "attempts", case)
    before = _sentinel_state(dest)

    if before is not None:
        rec.update(outcome="REJECTED_PREEXISTING_DESTINATION", launched=False,
                   attempt_dir=dest, preexisting=before,
                   preserved_sentinel=before,
                   sentinel_preservation=sentinel_preserved(before, _sentinel_state(dest)),
                   control_completion="CLEAN",
                   reason="attempt destination already exists; refusing to launch over another "
                          "attempt's evidence. No marker is written, the existing bytes are left "
                          "exactly as found, and this is a recorded nonlaunch outcome rather than "
                          "an arm result")
        return rec

    if budget_remaining is None or budget_remaining <= cap:
        rec.update(outcome="REJECTED_INSUFFICIENT_CAP", launched=False,
                   attempt_dir=dest, cap_s=cap, remaining_s=budget_remaining,
                   control_completion="CLEAN",
                   reason="remaining budget does not cover this arm's cap; refusing to start work "
                          "that cannot finish inside the original allocation. No marker, no "
                          "partial attempt directory, and no new deadline")
        return rec

    if launcher is None:
        rec.update(outcome="NOT_RUN", launched=False, attempt_dir=dest,
                   control_completion="CLEAN",
                   reason="no launcher supplied; this allocation authorizes zero native arms and "
                          "nothing is launched. NOT_RUN is the honest value and it is not a "
                          "negative activation result")
        return rec

    # R10-CI-008. One owner creates the attempt directory: this function. The
    # launcher is handed the directory that was claimed here, so there is no
    # second creator and no window in which two of them disagree about who owns
    # the destination. The seam is an ordinary parameter, which is what makes a
    # local lifecycle scenario possible without inventing a native collector or
    # acquiring authority to call a model.
    try:
        os.makedirs(dest)          # deliberately not exist_ok: the claim is the check
    except FileExistsError:
        rec.update(outcome="REJECTED_PREEXISTING_DESTINATION", launched=False,
                   attempt_dir=dest, control_completion="CLEAN",
                   reason="destination appeared between the inspection and the claim; refusing")
        return rec

    rec.update(launched=True, attempt_dir=dest, marker_written=True,
               outcome="LAUNCHED",
               reason="destination claimed exclusively; the arm's own activation observation "
                      "decides PASS or FAIL, and this marker records only that it was admitted")
    # The prompt actually handed over is the verified selected bytes, and it is
    # recorded beside the result so activation reads the same identity the
    # launch used.
    rec["launched_prompt_sha256"] = launched_sha
    rec["launcher_result"] = launcher(case, selector, dest, launch_prompt)
    rec["activation_inputs"] = {"attempt_dir": dest, "case": case,
                                "row_id": fixed["row_id"],
                                "should_trigger": fixed["should_trigger"],
                                "query_sha256": fixed["query_sha256"]}
    return rec


def nonlaunch_is_not_a_result(rec):
    """Guard for the gate: a nonlaunch never becomes an arm outcome."""
    return {"case": rec.get("case"),
            "counts_as_arm_result": rec.get("outcome") not in NONLAUNCH + ("NOT_RUN",),
            "creates_admission_marker": bool(rec.get("marker_written")),
            "note": "An ordinary refusal to launch is a truthful control outcome. It is neither a "
                    "successful arm nor evidence about the candidate, and it must not satisfy a "
                    "downstream prerequisite."}
