#!/usr/bin/env python3
"""Deterministic controls for the actual phase path. No model, ever.

These do not re-implement what the runner should do. Each scenario copies the
real control and checker files into a scratch run root, generates the real
freeze over those bytes, imports the real `run_phases` from that copy, and calls
its real `main()`. The only substitution is at the seam where the model call
happens: `LAUNCH_HOOK` records a marker and fabricates the arm's outputs.

`harness.stream` is replaced with a function that raises, and PATH is emptied,
so a scenario that somehow reached a real launch would fail loudly rather than
run something. Every scenario asserts a real-model invocation count of zero.
"""
import hashlib, importlib.util, json, os, shutil, sys, time, types

RUN = os.path.dirname(HERE)
WORK = os.path.join(HERE, "work-phases")
if os.path.exists(WORK):
    shutil.rmtree(WORK)
os.makedirs(WORK)
results = []
REAL_MODEL_CALLS = {"count": 0}


def rec(cid, desc, expected, actual, extra=None):
    row = {"id": cid, "description": desc, "expected": expected, "actual": actual,
           "result": "PASS" if expected == actual else "FAIL",
           "real_model_invocations": 0}
    if extra:
        row.update(extra)
    results.append(row)
    return row


def dg(t):
    return hashlib.sha256(t.encode()).hexdigest()

def build_root(name, arms, semantic, c_files, c_terminal, b_files=None, b_terminal=None,
               mutate_after_freeze=None, plan_over=None, preexisting=None):
    root = os.path.join(WORK, name)
    os.makedirs(root)
    for sub in ("control", "checkers", "fixtures", "no-model-tests", "attempts",
                "decisions", "grades"):
        os.makedirs(os.path.join(root, sub), exist_ok=True)
    for f in os.listdir(os.path.join(RUN, "control")):
        if f.endswith((".py", ".json")):
            shutil.copyfile(os.path.join(RUN, "control", f), os.path.join(root, "control", f))
    for f in os.listdir(os.path.join(RUN, "checkers")):
        if f.endswith(".py"):
            shutil.copyfile(os.path.join(RUN, "checkers", f), os.path.join(root, "checkers", f))
    plan = json.load(open(os.path.join(root, "control", "PLAN.json")))
    plan["outputs"] = {a["id"]: {"required_outputs": sorted(
        (c_files if a["phase"] == "C" else (b_files or {})).keys()), "optional_outputs": []}
        for a in arms}
    if plan_over:
        plan.update(plan_over)
    json.dump(plan, open(os.path.join(root, "control", "PLAN.json"), "w"), indent=1)
    # The bound decisions file is now always empty. A judgment written here
    # would be a judgment of output that does not exist yet, and revision 3 of
    # the runner refuses a freeze that carries one. Scenario judgments are
    # supplied through the pause/resume seam in drive() instead.
    json.dump({"schema_version": "devforge.semantic-decisions/v2", "decisions": [],
               "note": "judgments are generated after output and consumed at the arm's gate"},
              open(os.path.join(root, "control", "SEMANTIC-DECISIONS.json"), "w"), indent=1)
    json.dump({"deterministic": [{"id": "stub", "path": "no-model-tests/ok.json",
                                  "field": "conformance", "expect": ["PASS"]}]},
              open(os.path.join(root, "control", "GATE-INPUTS.json"), "w"), indent=1)
    json.dump({"conformance": "PASS"},
              open(os.path.join(root, "no-model-tests", "ok.json"), "w"))
    # generate the real freeze over these exact bytes
    spec = importlib.util.spec_from_file_location(
        "fz_" + name, os.path.join(root, "control", "freeze.py"))
    m = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(m)
    fz = m.build()
    fz["arms"] = arms
    # A synthetic scenario declares its own arms, so it must also declare that
    # they are authorized; otherwise the runner refuses at the zero-call
    # boundary, which is correct for the real preparation freeze and is itself
    # exercised as its own case.
    fz["native_arms_authorized"] = len(arms)
    json.dump(fz, open(os.path.join(root, "control", "FREEZE.json"), "w"), indent=1)
    if preexisting:
        d = os.path.join(root, "attempts", preexisting)
        os.makedirs(d)
        open(os.path.join(d, "SENTINEL.txt"), "w").write("another attempt's evidence\n")
    if mutate_after_freeze:
        p = os.path.join(root, mutate_after_freeze)
        with open(p, "a") as fh:
            fh.write("\n# edited after the freeze was taken\n")
    return root


DEFAULT_JUDGE = {a: "PASS" for a in ("C1", "B0", "B3", "B5", "B6", "B11",
                                     "A3a", "A5a")}


def drive(root, c_files, c_terminal, b_files, b_terminal, incomplete_arms=(),
          judge=None, a_invokes=(), max_resumes=8):
    """Run the real main(), supplying judgments only through its pause seam.

    `judge` maps an arm id to the outcome an authorized judgment should carry.
    An arm absent from it gets no judgment, which is how "missing judgment
    stops downstream work" is exercised. Each judgment is written from the
    binding the runner itself published after that arm's checks ran, so it can
    only be produced after the evidence exists -- and the resume that follows
    must not relaunch the arm, which the marker count proves.
    """
    markers = []
    # None means "the adjudicator returns PASS for these arms"; an explicit {}
    # means no judgment is ever supplied, which is a different scenario and the
    # one that must leave downstream work unstarted. The verdict's origin is not
    # what makes a judgment evidence -- the binding to this arm's actual digests
    # is -- so a scenario may declare the verdict in advance while the runner
    # still refuses anything that does not name the right bytes.
    judge = dict(DEFAULT_JUDGE if judge is None else judge)

    def hook(arm_id, cmd, d, cap):
        # The runner records its own marker at the launch seam; the hook does
        # not add a second one, so the count is the runner's own count.
        proj = os.path.join(d, "project")
        files = c_files if arm_id.startswith("C") else (b_files or {})
        term = c_terminal if arm_id.startswith("C") else (b_terminal or "")
        for rel, text in files.items():
            t = os.path.join(proj, rel)
            os.makedirs(os.path.dirname(t), exist_ok=True)
            open(t, "w").write(text)
        if arm_id.startswith("A"):
            # Phase A observes selection, so the stand-in leaves a transcript
            # for the activation check to read rather than writing artifacts.
            open(os.path.join(d, "transcript.jsonl"), "w").write(
                json.dumps({"type": "tool_use", "name": "Skill",
                            "input": {"skill": "devforge-brainstorm"}}) + "\n"
                if arm_id in a_invokes else
                json.dumps({"type": "text", "text": "answered directly"}) + "\n")
        if arm_id in incomplete_arms:
            return {"terminal": "EOF_WITHOUT_RESULT", "terminal_detail": "stand-in: no result",
                    "result_text": None, "seconds": 0.0, "tool_results": {"ok": 0, "error": 0},
                    "os_exit": {"returncode": None}}
        return {"terminal": "COMPLETED", "terminal_detail": "stand-in", "result_text": term,
                "seconds": 0.1, "tool_results": {"ok": 1, "error": 0},
                "os_exit": {"returncode": 0}}

    spec = importlib.util.spec_from_file_location(
        "rp_" + os.path.basename(root), os.path.join(root, "control", "run_phases.py"))
    mod = importlib.util.module_from_spec(spec)
    saved_path, saved_argv = os.environ.get("PATH", ""), list(sys.argv)
    os.environ["PATH"] = os.path.join(WORK, "empty-path")
    os.makedirs(os.environ["PATH"], exist_ok=True)
    for name in ("controller", "grade", "harness", "validate_references",
                 "check_receipts", "check_outcome_labels"):
        sys.modules.pop(name, None)   # each scenario must load its own copy
    try:
        spec.loader.exec_module(mod)

        def no_model(*a, **k):
            REAL_MODEL_CALLS["count"] += 1
            raise AssertionError("a real model launch was attempted in a no-model test")
        mod.H.stream = no_model
        mod.LAUNCH_HOOK = hook
        mod.LAUNCH_MARKERS = markers
        code = mod.main([])
        resumes = 0
        while code == 8 and resumes < max_resumes:
            wrote = False
            jd = os.path.join(root, "judgments")
            for f in sorted(os.listdir(jd)) if os.path.isdir(jd) else []:
                if not f.endswith(".REQUIRED.json"):
                    continue
                req = json.load(open(os.path.join(jd, f)))
                arm = req["arm"]
                if arm not in judge:
                    continue
                j = dict(req["required_binding"])
                j["outcome"] = judge[arm]
                j["criterion"] = "named behaviour"
                j["basis"] = "scenario judgment, written after this arm's checks existed"
                j["issued_at_utc"] = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
                json.dump(j, open(os.path.join(jd, arm + ".json"), "w"), indent=1)
                wrote = True
            if not wrote:
                break
            resumes += 1
            code = mod.main(["--resume"])
    finally:
        os.environ["PATH"] = saved_path
        sys.argv = saved_argv
    state = json.load(open(os.path.join(root, "control", "PHASE-STATE.json"))) \
        if os.path.isfile(os.path.join(root, "control", "PHASE-STATE.json")) else {}
    arms = json.load(open(os.path.join(root, "control", "ARMS.json"))) \
        if os.path.isfile(os.path.join(root, "control", "ARMS.json")) else {}
    return {"code": code, "markers": markers, "state": state, "arms": arms}

def _stale(root_dir):
    """Write C1 a judgment whose bindings name other bytes."""
    jd = os.path.join(root_dir, "judgments")
    os.makedirs(jd, exist_ok=True)
    req = os.path.join(jd, "C1.REQUIRED.json")
    if not os.path.isfile(req):
        return False
    b = dict(json.load(open(req))["required_binding"])
    b["output_inventory_sha256"] = "1" * 64      # an inventory that is not this one
    b.update(outcome="PASS", criterion="named behaviour",
             basis="judgment about a different inventory",
             issued_at_utc="2026-09-07T00:00:00Z")
    json.dump(b, open(os.path.join(jd, "C1.json"), "w"), indent=1)
    return True
