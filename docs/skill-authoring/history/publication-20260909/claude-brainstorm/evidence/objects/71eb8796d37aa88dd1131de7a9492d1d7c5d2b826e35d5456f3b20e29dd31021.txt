#!/usr/bin/env python3
"""Admission predicates: preservation, prerequisites, observations, authority.

Fresh executable source for CC7-003, CC7-004, CC7-005 and CC7-011 (CHG-007,
CHG-008, CHG-009, CHG-015), authored against the selected r7 control baseline
(control/PLAN.json, SHA-256 002099d3...; control/controller.py:64-93). Nothing
here is executed by this allocation, and nothing here launches anything: these
are the decision functions the phase path calls, written so that an absent
observation refuses rather than passes.

The baseline's `Controller.gate` reads checker files and recorded decisions and
is right about both. What it has no opinion about is the four questions below,
each of which was answered somewhere else, loosely or not at all:

  preserve_check   CC7-003  a retained archive must still be there, and be the
                            bytes that were selected -- not merely a key that
                            survived in an inventory.
  prerequisites    CC7-004  C before B, C and B before A, checked before any
                            marker is written rather than after.
  observations     CC7-005  the runtime-owned observations the bridge requires,
                            with the ones this interface cannot supply deferred
                            by name and failing closed.
  authority_bind   CC7-011  new evidence binds the current selected allocation,
                            never the historical one it was derived from.
"""
import hashlib, json, os, re

OUTCOMES = ("PASS", "FAIL", "NOT_RUN", "COULD_NOT_RUN", "NOT_APPLICABLE")


def sha(path):
    h = hashlib.sha256()
    with open(path, "rb") as fh:
        for b in iter(lambda: fh.read(1 << 16), b""):
            h.update(b)
    return h.hexdigest()


# --- CC7-003 / CHG-007: preservation is retained bytes ------------------------

def preserve_check(selected, run_root):
    """Every `preserved_outputs` path must hold its selected original bytes now.

    `selected` maps a project-relative path to the digest that was selected for
    it -- the original's digest, taken before the write that superseded it.

    The three states that must not collapse into one another:

      retained   the file is there and hashes to the selected original   PASS
      changed    the file is there and hashes to something else          FAIL
      vanished   the file is not there at the moment of judgment         FAIL

    The inventory is built from observed bytes, so an entry can outlive the file
    it describes -- the judgment pause is long enough for a later step to remove
    an archive, and a surviving inventory key then reads exactly like a surviving
    file. The digest is recomputed here, at judgment time, against the selected
    original: an exclusion claim is only as good as the bytes behind it
    (PLAN.json:26; bridge:21).
    """
    rows, ok = [], True
    for rel, original in sorted(selected.items()):
        path = rel if os.path.isabs(rel) else os.path.join(run_root, rel)
        row = {"preserved_output": rel, "selected_original_sha256": original}
        if not os.path.isfile(path):
            row.update(state="VANISHED", outcome="FAIL",
                       reason="selected preserved output is not present at judgment time; a "
                              "surviving inventory key is not surviving bytes")
            ok = False
        else:
            got = sha(path)
            row["actual_sha256"] = got
            if got != original:
                row.update(state="CHANGED", outcome="FAIL",
                           reason="preserved output no longer holds the bytes that were selected; "
                                  "exclusion requires exact digest identity, and this comparison "
                                  "is the record of it")
                ok = False
            else:
                row.update(state="RETAINED", outcome="PASS",
                           reason="live bytes recomputed at judgment time equal the selected "
                                  "original digest")
        rows.append(row)
    return {"check": "preservation", "rows": rows,
            "outcome": "PASS" if ok else "FAIL",
            "comparison_recorded": True,
            "note": "Required outputs are inspected regardless of retention class. Preservation "
                    "is a statement about bytes that are still reachable, so it is decided from "
                    "the file, never from the inventory that described it."}


# --- CC7-004 / CHG-008: C before B, C and B before A --------------------------

# The fixed plan names these arms and this order (PLAN.json:43-56; bridge:46).
REQUIRED_C = ("C1",)
REQUIRED_B = ("B5", "B11")
REQUIRED_BEFORE_A = REQUIRED_C + REQUIRED_B
A_ARMS = ("A3a", "A5a")


def prerequisites(arm, completed, diagnostic_scope=None):
    """May `arm` be admitted, given what has actually completed?

    `completed` maps an arm ID to its recorded result: it must carry a checked
    mechanical outcome and a bound post-output judgment. An arm that ran but was
    never checked, or was checked but never judged, has not satisfied anything.

    The refusal happens here, before a marker is written. Revision 2's ordering
    was enforced by the sequence of calls in the runner, which is an ordering
    only as long as nobody calls them out of order; a refusal after the fact
    leaves an admission marker behind that says the arm was admitted.

    `diagnostic_scope` is the explicitly authorized C-only mode. It permits the
    C arm and refuses everything downstream, which is the point of it: a
    diagnostic is a scope, not a partial admission that later counts.
    """
    def satisfied(name):
        """Has `name` produced what the fixed plan actually asks of it?

        R9-CI-008: revision 2 asked one question of every prerequisite -- a
        checked mechanical PASS plus a bound post-output judgment. That is the
        right question for C and B, whose arms produce artifacts that a semantic
        judgment is written about. It is the wrong question for A3a, which
        produces no artifact map at all (PLAN.json:90-110): its result is an
        activation observation. Requiring a C/B-style semantic judgment of it
        meant a perfectly valid A3a activation could never satisfy A5a's guard
        without someone inventing judgment fields the fixed plan does not
        define, and inventing them is the failure mode, not the fix.

        So A3a is judged on the evidence A3a actually has: it completed, and its
        own activation observation was recorded and passed. That is completion
        and activation sequencing, and it is deliberately not a semantic
        judgment -- A5a consumes A3a's *result*, not an opinion about it.
        """
        rec = (completed or {}).get(name) or {}
        if name in A_ARMS:
            act = rec.get("activation") or {}
            return (rec.get("native") == "PASS"
                    and act.get("outcome") == "PASS"
                    and act.get("state") not in ("UNUSABLE", "NO_TRANSCRIPT", "INCOMPLETE"))
        return (rec.get("mechanical") == "PASS"
                and rec.get("judgment") == "PASS"
                and bool(rec.get("judgment_bound_to")))

    out = {"arm": arm, "requires": [], "unsatisfied": [], "marker_written": False}

    if diagnostic_scope:
        out["scope"] = "C_ONLY_DIAGNOSTIC"
        if arm in REQUIRED_C:
            out.update(admit=True, outcome="PASS",
                       reason="explicitly authorized C-only diagnostic scope admits the C arm")
            return out
        out.update(admit=False, outcome="FAIL", requires=list(REQUIRED_C),
                   reason="an authorized C-only diagnostic scope admits nothing downstream; its "
                          "results are diagnostics and do not stand in for B or A admission")
        return out

    if arm in REQUIRED_C:
        out.update(admit=True, outcome="PASS", reason="C has no prerequisite arm")
        return out
    if arm in REQUIRED_B:
        need = list(REQUIRED_C) + [b for b in REQUIRED_B if b != arm and _ordered_before(b, arm)]
    elif arm in A_ARMS:
        need = list(REQUIRED_BEFORE_A)
        if arm == "A5a":
            # A5a follows A3a. The prerequisite is A3a's completion and
            # activation result, evaluated by `satisfied` under the A-arm rule
            # above -- not a semantic judgment A3a was never asked to produce.
            need = need + ["A3a"]
            out["a3a_prerequisite_kind"] = "completion_and_activation_evidence"
    else:
        out.update(admit=False, outcome="FAIL",
                   reason="unknown arm %r is not in the fixed plan" % (arm,))
        return out

    out["requires"] = need
    out["unsatisfied"] = [n for n in need if not satisfied(n)]
    if out["unsatisfied"]:
        out.update(admit=False, outcome="FAIL",
                   reason="refusing admission: %s has not completed with a checked result and a "
                          "bound judgment. No admission marker is written for a refused arm"
                          % ", ".join(out["unsatisfied"]))
        return out
    out.update(admit=True, outcome="PASS",
               reason="every required prerequisite completed, was checked and carries a bound "
                      "post-output judgment")
    return out


def _ordered_before(candidate, arm):
    order = list(REQUIRED_B)
    return order.index(candidate) < order.index(arm)


# --- CC7-005 / CHG-009: runtime-owned observations ----------------------------

# The bridge (lines 40-46) requires these as independent observations. Each is a
# distinct thing seen, not a property inferred from a valid artifact: worker
# text, static presence and exported final hashes are none of them.
RUNTIME_OBSERVATIONS = {
    "ordered_checkpoints": "runtime-recorded phase order and per-phase checkpoint acceptance",
    "stop_and_correction": "actual Stop callback and bounded-correction events",
    "receipt_publication": "runtime publication, readback and current-target verification of the "
                           "receipt, with its returned locator and digest",
    "installed_resource_loading": "observed consultation of the installed SKILL/reference/template "
                                  "bytes",
    "rendered_delivery": "the response actually rendered to a client; a successful socket write is "
                         "not rendering",
}

# Observations the selected v1 interface exposes no collector for. They are
# deferred by name and they fail closed: a deferred observation blocks the gate
# that depends on it, exactly as a missing one does. What must never happen is
# the third option -- inventing a collector API, or letting a worker's own text
# stand in (REMEDIATION-SPECIFICATION.md; CC7-005).
ROOT_DEPENDENCY_DEFERRED = ("stop_and_correction", "receipt_publication", "rendered_delivery")


HEX64 = re.compile(r"\A[0-9a-f]{64}\Z")

# A runtime-owned source is one the protected runtime writes. Worker prose, a
# grader's own summary and an exported final hash are not observations of the
# runtime; they are what the observations were supposed to be independent of.
# The selected v1 interface exposes these mechanical records (see
# references/managed-runtime.md and the selected phase_state/delivery_core/
# hook_protocol bytes), and nothing else.
RUNTIME_OWNED_SOURCES = (
    "runtime.phase_state",       # ordered checkpoint acceptance, protected correction/accepted
    "runtime.delivery_core",     # verified receipt mechanical result
    "runtime.hook_protocol",     # resume/stop protocol records
)

SUCCESSFUL_OUTCOMES = ("PASS", "OBSERVED", "SUCCESS")

# R10-CI-006. The two implementable classes accepted any record whose source
# string was `runtime.phase_state`, whose task string matched, and whose digest
# was 64 hex characters. None of that establishes what the class actually
# claims: an ordered checkpoint sequence is a *sequence*, and installed-resource
# consultation is a statement about *which resource* was read. A record shaped
# like a runtime record is not a runtime observation.
#
# Each class therefore declares the source that owns it and the fields whose
# presence and shape can establish it. A record that cannot carry the claim is
# refused as unsupported for that class rather than accepted for looking right.
CLASS_CONTRACT = {
    "ordered_checkpoints": {
        "sources": ("runtime.phase_state",),
        "requires": ("checkpoints",),
        "establishes": "an ordered, per-phase accepted checkpoint sequence",
        "shape": "checkpoints must be a non-empty list of {phase, accepted} in recorded order",
    },
    "installed_resource_loading": {
        "sources": ("runtime.phase_state", "runtime.delivery_core"),
        "requires": ("resource_path", "resource_sha256"),
        "establishes": "consultation of a named installed resource, by path and digest",
        "shape": "resource_path names the installed file and resource_sha256 is its digest",
    },
    "stop_and_correction": {
        "sources": ("runtime.hook_protocol",),
        "requires": ("events",),
        "establishes": "actual Stop and bounded-correction callback events",
        "shape": "events must be a non-empty list of recorded callback events",
    },
    "receipt_publication": {
        "sources": ("runtime.delivery_core",),
        "requires": ("published_locator", "readback_sha256"),
        "establishes": "publication, readback and current verification of the receipt",
        "shape": "published_locator and readback_sha256 come from the runtime's own publication",
    },
    "rendered_delivery": {
        "sources": ("runtime.delivery_core",),
        "requires": ("rendered_bytes_sha256",),
        "establishes": "the response actually rendered to a client",
        "shape": "a successful socket write is not rendering",
    },
}


def _checkpoints_ordered(rows):
    """A checkpoint record establishes ordering only if it carries one."""
    if not isinstance(rows, list) or not rows:
        return False, "checkpoints is empty or not a list, so it records no sequence"
    for r in rows:
        if not isinstance(r, dict) or "phase" not in r or "accepted" not in r:
            return False, "a checkpoint row lacks phase/accepted, so no order can be read from it"
        if r.get("accepted") is not True:
            return False, "checkpoint for phase %r is not recorded accepted" % r.get("phase")
    return True, ""


def _evidence_bytes_match(rec):
    """Recompute the digest of the selected evidence file the record names.

    An observation that names no reachable evidence file cannot be checked
    against anything, so it is refused: `evidence_path` is required for a record
    that claims to have observed bytes. This is what separates "the runtime
    recorded these bytes" from "somebody typed a plausible digest".
    """
    path = rec.get("evidence_path")
    if not path or not os.path.isfile(path):
        return False
    return sha(path) == rec.get("observed_bytes_sha256")


def observations(observed, required=None, task_identity=None):
    """Fail-closed consumer for the runtime observation record.

    R9-CI-003: revision 2 accepted any object carrying two truthy keys, so
    ``{"source": "worker text", "observed_bytes_sha256": "x", "outcome": "FAIL"}``
    became PASS -- an unsuccessful outcome, a malformed digest and a source the
    runtime does not own, all three waved through. Worse, a deferred class was
    only deferred when *nothing* was supplied for it: handing it a dictionary
    dropped it into the ordinary branch, so the deferral that was supposed to
    fail closed could be bypassed by inventing a row for it.

    Every one of those is checked here, and each has its own reason so a reader
    can tell which thing was wrong:

      unknown name        not one of the five fixed classes        -> FAIL
      narrowed coverage   caller asked for a subset of the five    -> FAIL
      deferred            no collector exists in selected v1       -> COULD_NOT_RUN, blocks
      missing             nothing recorded                          -> FAIL
      unusable            not an object, or no source/digest        -> FAIL
      bad provenance      source the runtime does not own          -> FAIL
      bad digest syntax   not a whole lowercase 64-hex token       -> FAIL
      wrong task          bound to some other task identity        -> FAIL
      unsuccessful        the observation itself reports failure   -> FAIL

    Nothing in this function can return PASS on the strength of an artifact
    existing, and no path returns PASS for a deferred class.
    """
    rows, blocking = [], []
    requested = tuple(required) if required else tuple(RUNTIME_OBSERVATIONS)

    # The five classes are fixed. A caller that asks for fewer is narrowing the
    # required set after the criteria were frozen, which is the move CC7-005
    # exists to refuse -- so the check runs over the full set regardless, and
    # the narrowing itself is recorded as a failure.
    narrowed = [n for n in RUNTIME_OBSERVATIONS if n not in requested]
    unknown = [n for n in requested if n not in RUNTIME_OBSERVATIONS]
    for name in sorted(RUNTIME_OBSERVATIONS):
        row = {"observation": name, "requires": RUNTIME_OBSERVATIONS[name]}
        rec = (observed or {}).get(name)

        if name in ROOT_DEPENDENCY_DEFERRED:
            # Deferred first, and unconditionally. Whether a row was supplied
            # changes nothing: the selected v1 interface exposes no collector
            # for this observation, so no supplied dictionary can be one.
            row.update(outcome="COULD_NOT_RUN", state="ROOT_DEPENDENCY_DEFERRED",
                       blocks_dependent_admission=True,
                       supplied_row_ignored=isinstance(rec, dict),
                       reason="the selected v1 interface exposes no collector for this "
                              "observation. It is deferred by name, it is not satisfied, and it "
                              "blocks the gate that depends on it. A dictionary supplied for it is "
                              "not a collector and is ignored rather than accepted; do not "
                              "substitute worker text, static artifact presence or an exported "
                              "final hash")
            blocking.append(name)
            rows.append(row)
            continue

        if not isinstance(rec, dict):
            row.update(outcome="FAIL", state="MISSING", blocks_dependent_admission=True,
                       reason="no observation recorded; a valid artifact is not evidence that the "
                              "runtime did the thing that produces it")
            blocking.append(name)
            rows.append(row)
            continue

        src, dig = rec.get("source"), rec.get("observed_bytes_sha256")
        # R10-CI-006. `rec.get("outcome", "PASS")` let a record that never said
        # how it went be read as having gone well. An observation with no
        # recorded outcome has not reported success.
        outcome = rec.get("outcome")
        contract = CLASS_CONTRACT.get(name, {})
        if outcome is None:
            row.update(outcome="FAIL", state="NO_OUTCOME_RECORDED",
                       blocks_dependent_admission=True,
                       reason="the record states no outcome; a missing outcome is not a "
                              "successful one and does not default to PASS")
            blocking.append(name)
            rows.append(row)
            continue
        if not src or not dig:
            row.update(outcome="FAIL", state="UNUSABLE", blocks_dependent_admission=True,
                       reason="an observation must name the runtime-owned source it came from and "
                              "the exact bytes observed; a bare assertion is not an observation")
        elif src not in RUNTIME_OWNED_SOURCES:
            row.update(outcome="FAIL", state="WRONG_PROVENANCE", source=src,
                       blocks_dependent_admission=True,
                       accepted_sources=list(RUNTIME_OWNED_SOURCES),
                       reason="source %r is not a runtime-owned record. Worker text, a grader "
                              "summary and an exported final hash are exactly what these "
                              "observations are required to be independent of" % (src,))
        elif not (isinstance(dig, str) and HEX64.match(dig)):
            row.update(outcome="FAIL", state="BAD_DIGEST_SYNTAX", observed_bytes_sha256=dig,
                       blocks_dependent_admission=True,
                       reason="observed_bytes_sha256 %r is not a whole lowercase 64-character "
                              "hex digest, so it identifies no bytes" % (dig,))
        elif task_identity is not None and rec.get("task_id") != task_identity:
            row.update(outcome="FAIL", state="WRONG_TASK_IDENTITY",
                       expected_task_id=task_identity, task_id=rec.get("task_id"),
                       blocks_dependent_admission=True,
                       reason="observation is bound to task %r, not to the admitted task %r; an "
                              "observation of other work is not an observation of this one"
                              % (rec.get("task_id"), task_identity))
        elif src not in contract.get("sources", ()):
            row.update(outcome="FAIL", state="WRONG_SOURCE_FOR_CLASS", source=src,
                       blocks_dependent_admission=True,
                       accepted_sources_for_class=list(contract.get("sources", ())),
                       reason="source %r does not own this observation class. %s is established "
                              "by %s, and a record from elsewhere cannot carry that claim"
                              % (src, name, " or ".join(contract.get("sources", ())) or "nothing"))
        elif [k for k in contract.get("requires", ()) if k not in rec]:
            missing_fields = [k for k in contract.get("requires", ()) if k not in rec]
            row.update(outcome="FAIL", state="UNSUPPORTED_FOR_CLASS",
                       blocks_dependent_admission=True, missing_fields=missing_fields,
                       reason="the record lacks %s, so it cannot establish %s (%s)"
                              % (", ".join(missing_fields), name, contract.get("establishes", "")))
        elif name == "ordered_checkpoints" and not _checkpoints_ordered(rec.get("checkpoints"))[0]:
            row.update(outcome="FAIL", state="NO_ORDER_ESTABLISHED",
                       blocks_dependent_admission=True,
                       reason=_checkpoints_ordered(rec.get("checkpoints"))[1])
        elif not _evidence_bytes_match(rec):
            row.update(outcome="FAIL", state="ACTUAL_BYTES_MISMATCH",
                       blocks_dependent_admission=True,
                       evidence_path=rec.get("evidence_path"),
                       reason="observed_bytes_sha256 does not equal the digest of the selected "
                              "evidence file named by evidence_path, so the record describes "
                              "bytes nobody can point at")
        elif outcome not in SUCCESSFUL_OUTCOMES:
            row.update(outcome="FAIL", state="UNSUCCESSFUL", recorded_outcome=outcome,
                       blocks_dependent_admission=True,
                       reason="the observation itself reports %r. An observation that records the "
                              "runtime failing to do the thing is not evidence that it did it"
                              % (outcome,))
        else:
            row.update(outcome="PASS", state="OBSERVED", source=src,
                       observed_bytes_sha256=dig, recorded_outcome=outcome,
                       blocks_dependent_admission=False)
        if row["outcome"] != "PASS":
            blocking.append(name)
        rows.append(row)

    if narrowed:
        rows.append({"observation": "__coverage__", "outcome": "FAIL", "state": "NARROWED",
                     "omitted": narrowed, "blocks_dependent_admission": True,
                     "reason": "the caller asked for a subset of the five fixed observation "
                               "classes. A caller cannot narrow or replace the required set after "
                               "the criteria were frozen; the omitted classes were checked anyway "
                               "and the narrowing is itself a failure"})
        blocking.append("__coverage__")
    if unknown:
        rows.append({"observation": "__coverage__", "outcome": "FAIL", "state": "UNKNOWN_CLASS",
                     "unknown": unknown, "blocks_dependent_admission": True,
                     "reason": "requested observation class(es) %s are not among the five fixed "
                               "classes" % ", ".join(sorted(unknown))})
        blocking.append("__coverage__")

    return {"check": "runtime_observations", "rows": rows,
            "required_classes": sorted(RUNTIME_OBSERVATIONS),
            "blocking": blocking,
            "deferred": [n for n in blocking if n in ROOT_DEPENDENCY_DEFERRED],
            "outcome": "PASS" if not blocking else "FAIL",
            "native_admission": "BLOCKED" if blocking else "NOT_BLOCKED_BY_THIS_CHECK",
            "note": "This check never returns PASS on the strength of an artifact existing. "
                    "Deferred entries are honest gaps in the interface and are reported as gaps; "
                    "they are not a licence to skip the observations that are implementable, and "
                    "no dictionary supplied for a deferred class can satisfy it.",
            "not_established_here": ["native completion", "provider stream EOF",
                                     "owned process cleanup", "process caps"]}


# --- CC7-011 / CHG-015: new evidence binds the current allocation -------------

AUTHORITY_KEYS = ("session", "revision", "path", "sha256")


def authority_bind(selected, emitted):
    """New records must carry the selected *current* allocation, exactly.

    Historical clocks, receipts and authority records are inputs. Carrying one
    into a new record is how a finished allocation's identity ends up asserted
    about work it never covered (SESSION-001-r7.md:36,46,58). So the emitted
    identity is compared field by field against the selected one, and a
    mismatch is rejected rather than reconciled toward either side.
    """
    rows, bad = [], False
    for key in AUTHORITY_KEYS:
        want, got = selected.get(key), (emitted or {}).get(key)
        row = {"field": key, "selected": want, "emitted": got,
               "outcome": "PASS" if got == want and got is not None else "FAIL"}
        if row["outcome"] == "FAIL":
            bad = True
            row["reason"] = ("emitted value does not equal the selected current allocation"
                             if got is not None else "no value emitted for this field")
        rows.append(row)
    return {"check": "authority_binding", "rows": rows,
            "selected": dict(selected), "emitted": dict(emitted or {}),
            "outcome": "FAIL" if bad else "PASS",
            "note": "A historical authority is an input, never the identity of new work. Rejection "
                    "is the outcome; rewriting the record to match is not available here."}
