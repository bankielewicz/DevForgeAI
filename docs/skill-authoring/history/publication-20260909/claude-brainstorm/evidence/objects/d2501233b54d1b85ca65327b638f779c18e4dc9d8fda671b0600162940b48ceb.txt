"""Deterministic controls for the three checkers and the inventory. No model.

Each case fixes an expected outcome for a stated input, including every mutant
the independent review used to expose the previous revision. Where a defect was
reported, the fixture that exposed it is kept and its expectation is the correct
one -- so a regression re-surfaces as a failing case rather than as silence.
"""

results = []


def rec(group, cid, desc, expected, actual, extra=None):
    row = {"group": group, "id": cid, "description": desc, "expected": expected,
           "actual": actual, "result": "PASS" if expected == actual else "FAIL"}
    if extra:
        row.update(extra)
    results.append(row)
    return row


def w(path, text):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    open(path, "w").write(text)
    return path


def dg(text):
    return hashlib.sha256(text.encode()).hexdigest()

def ref_case(cid, desc, ref_block, target_text, expected, missing="[]"):
    d = os.path.join(WORK, "ref-" + cid)
    proj = os.path.join(d, "project")
    w(os.path.join(proj, "target.md"), target_text)
    w(os.path.join(proj, "OUTPUT.md"),
      "---\nschema_version: \"devforge.artifact/v1\"\nartifact_id: \"OUT-001\"\nrevision: 1\n"
      + ref_block + "\nmissing_inputs: " + missing + "\n---\n\n# Output\n")
    out = V.run(proj, {"project": proj}, ["OUTPUT.md"])
    refs = out["reference_conformance"]
    got = refs[0]["outcome"] if refs else ("PARSE_ERROR" if out["parse_errors"] else "NO_REF")
    reason = (refs[0]["reason"] if refs else
              (out["parse_errors"][0]["error"] if out["parse_errors"] else ""))
    return rec("references", cid, desc, expected, got,
               {"reason": reason[:200],
                "parsed_sections": out["parsed_section_claim_count"]})


def evd(target_text, kind='"raw-file"', drop=()):
    """An `evidence` entry -- the field where a raw file legitimately lives.

    The revision-2 suite built this case with blk(), which emits `upstream`, so
    the supposedly valid raw-evidence control was actually blessing a raw-file
    override on an artifact-reference field (CP4-SC-004 / checker review). The
    positive control has to construct the field it claims to be testing.
    """
    rows = [("store", "project"), ("path", "target.md"),
            ("sha256", '"%s"' % dg(target_text))]
    body = "evidence:\n  - " + "\n    ".join("%s: %s" % (k, v) for k, v in rows
                                              if k not in drop)
    body += "\n"
    if kind:
        body += "    evidence_kind: %s\n" % kind
    return body


def sup(target_text, kind=None, drop=()):
    """A `supersedes` mapping -- an artifact-reference field, like upstream."""
    rows = [("artifact_id", '"SOURCE-001"'), ("revision", "1"), ("store", "project"),
            ("path", "target.md"), ("sha256", '"%s"' % dg(target_text))]
    body = "supersedes:\n  " + "\n  ".join("%s: %s" % (k, v) for k, v in rows
                                            if k not in drop) + "\n"
    if kind:
        body += "  evidence_kind: %s\n" % kind
    return body


def blk(target_text, sections=None, inline=False, drop=(), extra=""):
    rows = [("artifact_id", '"SOURCE-001"'), ("revision", "1"), ("store", "project"),
            ("path", "target.md"), ("sha256", '"%s"' % dg(target_text))]
    body = "upstream:\n  - " + "\n    ".join("%s: %s" % (k, v) for k, v in rows if k not in drop)
    body += "\n"
    if extra:
        body += "    " + extra + "\n"
    if sections is not None:
        if isinstance(sections, str):
            body += "    sections: %s\n" % sections
        elif inline:
            body += "    sections: [%s]\n" % ", ".join('"%s"' % s for s in sections)
        else:
            body += "    sections:\n" + "".join("      - %s\n" % s for s in sections)
    return body

MAPPING = {"accepted_forms": ["project_relative", "sandbox_absolute"],
           "record_shapes": ["locator_then_digest", "digest_then_locator",

def rcp_case(cid, desc, files, terminal, subject, expected, mapping=None, mutate=None):
    d = os.path.join(WORK, "rcp-" + cid)
    proj = os.path.join(d, "project")
    promised = []
    for rel, text in files.items():
        w(os.path.join(proj, rel), text)
        promised.append({"path": os.path.join(proj, rel), "label": rel})
    subs = {rel: dg(text) for rel, text in files.items()}
    term = terminal
    for rel, h in subs.items():
        term = term.replace("{H:%s}" % rel, h).replace("{SHORT:%s}" % rel, h[:8] + "..." + h[-4:])
        term = term.replace("{P:%s}" % rel, rel).replace("{S:%s}" % rel, "/work/arm/" + rel)
        term = term.replace("{HOST:%s}" % rel, os.path.join(proj, rel))
        term = term.replace("{OTHERROOT:%s}" % rel, "/some/unselected/root/" + rel)
    tp = w(os.path.join(d, "terminal.txt"), term)
    if mutate:
        for rel, extra in mutate.items():
            open(os.path.join(proj, rel), "a").write(extra)
    out = CR.evaluate({"terminal_source": tp, "locator_mapping": mapping or MAPPING,
                       "promised": promised})
    row = [r for r in out["receipts"] if r["receipt_for"] == subject][0]
    return rec("receipts", cid, desc, expected, row["outcome"],
               {"reason": row["reason"][:200], "association": row.get("association")})

# R10 discriminating cases (CC9-RR-001, CC9-RR-002, CC9-RR-003)
#
# These are wired to the revised checkers through the same `ref_case` / `rcp_case`
# helpers as every other case, so they resolve to `RUN/checkers` -- the freshly
# authored tree -- rather than to a baseline copy. CC9-RR-003 named exactly that
# gap: cases that describe the repaired behaviour but bind to the old code test
# the old code while appearing to cover the repair.
#
# Each case stages valid prerequisite bytes -- a real target, its real digest, a
# parseable envelope -- so the run reaches the predicate the case is named for.
# A case that fails early because its target is missing or its digest is wrong
# has not exercised disclosure classification or locator ownership, and would
# report the right outcome for the wrong reason.
# =====================================================================
